package com.voxelgame.engine;

import com.badlogic.gdx.Gdx;
import com.badlogic.gdx.audio.Music;
import com.badlogic.gdx.files.FileHandle;
import com.badlogic.gdx.utils.Disposable;

import java.io.File;
import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.stream.Stream;

/**
 * Hệ thống Quản lý Âm thanh & Nhạc nền (Background Music Manager) trong libGDX,
 * triển khai giao diện {@link Disposable}.
 *
 * Lớp này sử dụng đối tượng {@link Music} của libGDX (chạy trên OpenAL) để phát luồng (stream)
 * trực tiếp file MP3, hỗ trợ lặp vô hạn và điều chỉnh âm lượng mượt mà theo thời gian thực
 * mà không gây nghẽn luồng xử lý đồ họa (Render Thread).
 *
 * THỂ HIỆN CÁC TÍNH CHẤT OOP:
 * 1. Tính Trừu tượng (Abstraction):
 *    - Che giấu toàn bộ cơ chế giải mã âm thanh OpenAL đa nền tảng, quản lý bộ đệm streaming
 *      và luồng phát ngầm của libGDX đằng sau các phương thức giao tiếp đơn giản:
 *      {@link #startBackgroundMusic()}, {@link #setVolume(float)}, {@link #changeVolume(float)},
 *      và {@link #dispose()}.
 * 2. Tính Đóng gói (Encapsulation):
 *    - Trạng thái âm lượng {@code volume} [0.0f, 1.0f] và đối tượng phần cứng {@code backgroundMusic}
 *      được đóng gói an toàn ở phạm vi {@code private}. Các thao tác thay đổi âm lượng luôn được
 *      kẹp trong đoạn hợp lệ thông qua setter duy nhất {@link #setVolume(float)}.
 * 3. Quản lý Vòng đời Tài nguyên (Resource Lifecycle):
 *    - Triển khai {@link Disposable} để đồng bộ cơ chế giải phóng tài nguyên OpenAL cùng với
 *      toàn bộ hệ sinh thái libGDX của dự án.
 */
public class AudioManager implements Disposable {

    /** Mức âm lượng hiện tại trong đoạn [0.0f, 1.0f], mặc định 70% (0.70f) */
    private float volume = 0.70f;

    /** Cờ trạng thái đánh dấu nhạc nền đang được bật */
    private boolean running = false;

    /** Đối tượng quản lý stream âm thanh phần cứng OpenAL của libGDX */
    private Music backgroundMusic;

    /** Con trỏ tập tin đại diện cho file nhạc nền MP3 */
    private FileHandle musicFileHandle;

    /**
     * Khởi tạo AudioManager:
     * Tự động tìm kiếm file nhạc MP3 trong Classpath (`src/main/resources`), thư mục dự án
     * hoặc các thư mục con, sau đó khởi tạo đối tượng {@link Music}.
     */
    public AudioManager() {
        // Tự động tìm kiếm linh hoạt file nhạc "1-08. Minecraft.mp3"
        this.musicFileHandle = findMp3FileHandleFlexible(
                "1-08. Minecraft.mp3",
                "audio/1-08. Minecraft.mp3",
                "music/1-08. Minecraft.mp3",
                "minecraft.mp3",
                "bgm.mp3"
        );

        if (this.musicFileHandle != null) {
            try {
                this.backgroundMusic = Gdx.audio.newMusic(this.musicFileHandle);
                this.backgroundMusic.setLooping(true);
                this.backgroundMusic.setVolume(this.volume);
            } catch (Exception e) {
                System.err.println("[AudioManager] Không thể khởi tạo luồng âm thanh từ file: "
                        + this.musicFileHandle.path() + " - Lỗi: " + e.getMessage());
                this.backgroundMusic = null;
            }
        }
    }

    /**
     * Bắt đầu phát lặp lại (Loop) nhạc nền vô hạn trên luồng âm thanh OpenAL.
     * Nếu chưa nạp được nhạc, phương thức sẽ thử tìm kiếm lại một lần nữa trước khi báo lỗi.
     */
    public void startBackgroundMusic() {
        if (backgroundMusic == null) {
            // Thử quét lại tập tin phòng trường hợp file nhạc vừa được người dùng copy vào
            if (musicFileHandle == null) {
                musicFileHandle = findMp3FileHandleFlexible(
                        "1-08. Minecraft.mp3",
                        "audio/1-08. Minecraft.mp3",
                        "music/1-08. Minecraft.mp3",
                        "minecraft.mp3",
                        "bgm.mp3"
                );
            }
            if (musicFileHandle != null) {
                try {
                    backgroundMusic = Gdx.audio.newMusic(musicFileHandle);
                    backgroundMusic.setLooping(true);
                    backgroundMusic.setVolume(this.volume);
                } catch (Exception e) {
                    System.err.println("[AudioManager] Khởi tạo Music thất bại: " + e.getMessage());
                    backgroundMusic = null;
                }
            }
        }

        if (backgroundMusic == null) {
            System.err.println("[AudioManager] Không tìm thấy file nhạc '.mp3'. "
                    + "Hãy đặt file '1-08. Minecraft.mp3' vào thư mục 'src/main/resources/audio/' hoặc thư mục gốc của project.");
            return;
        }

        if (running && backgroundMusic.isPlaying()) {
            return;
        }

        try {
            backgroundMusic.setLooping(true);
            backgroundMusic.setVolume(this.volume);
            backgroundMusic.play();
            running = true;
            System.out.println("[AudioManager] Bắt đầu phát lặp nhạc nền thành công (Âm lượng: " + getVolumePercent() + "%).");
        } catch (Exception e) {
            System.err.println("[AudioManager] Lỗi khi phát nhạc nền: " + e.getMessage());
        }
    }

    /**
     * Thiết lập mức âm lượng nhạc nền theo thời gian thực (từ 0.0f = Tắt tiếng đến 1.0f = 100%).
     * Được gọi trực tiếp khi người chơi bấm nút [-]/[+] hoặc kéo thanh trượt Volume trong Pause Menu.
     *
     * @param newVolume giá trị âm lượng mới
     */
    public void setVolume(float newVolume) {
        this.volume = Math.max(0.0f, Math.min(1.0f, newVolume));
        if (backgroundMusic != null) {
            try {
                backgroundMusic.setVolume(this.volume);
            } catch (Exception ignored) {
            }
        }
    }

    /**
     * Tăng hoặc giảm âm lượng một khoảng {@code delta} (ví dụ: +0.10f hoặc -0.10f).
     *
     * @param delta mức thay đổi âm lượng
     */
    public void changeVolume(float delta) {
        setVolume(this.volume + delta);
    }

    /**
     * Lấy mức âm lượng hiện tại trong khoảng [0.0f, 1.0f].
     *
     * @return giá trị âm lượng dạng số thực
     */
    public float getVolume() {
        return volume;
    }

    /**
     * Lấy mức âm lượng hiện tại tính theo phần trăm nguyên (0 đến 100) để hiển thị trên Pause Menu.
     *
     * @return phần trăm âm lượng (0 - 100)
     */
    public int getVolumePercent() {
        return Math.round(volume * 100.0f);
    }

    /**
     * Kiểm tra trạng thái phát nhạc hiện tại.
     *
     * @return true nếu nhạc nền đang phát
     */
    public boolean isPlaying() {
        return backgroundMusic != null && backgroundMusic.isPlaying();
    }

    /**
     * Giải phóng toàn bộ tài nguyên âm thanh OpenAL khi đóng game,
     * triển khai phương thức của giao diện {@link Disposable}.
     */
    @Override
    public void dispose() {
        running = false;
        if (backgroundMusic != null) {
            try {
                if (backgroundMusic.isPlaying()) {
                    backgroundMusic.stop();
                }
                backgroundMusic.dispose();
            } catch (Exception ignored) {
            } finally {
                backgroundMusic = null;
            }
        }
    }

    /**
     * Phương thức tương thích ngược với giao diện Cleanable cũ, ủy quyền trực tiếp cho {@link #dispose()}.
     */
    public void cleanup() {
        dispose();
    }

    /**
     * Cơ chế tìm kiếm linh hoạt con trỏ file `.mp3` qua API {@link FileHandle} của libGDX:
     * 1. Tìm trong Internal/Classpath assets (`src/main/resources/`)
     * 2. Tìm trong thư mục cục bộ (Local working directory)
     * 3. Tìm theo các đường dẫn cơ sở thông dụng (`audio/`, `music/`, `src/main/resources/audio/`)
     * 4. Quét tự động cây thư mục tối đa 4 tầng để nhận diện bất kỳ file `.mp3` nào.
     *
     * @param candidateNames danh sách tên file hoặc đường dẫn ứng viên
     * @return {@link FileHandle} nếu tìm thấy, hoặc {@code null} nếu không tìm thấy
     */
    private FileHandle findMp3FileHandleFlexible(String... candidateNames) {
        String[] baseDirs = {
                "",
                "src/main/resources/",
                "src/main/resources/audio/",
                "audio/",
                "music/"
        };

        // 1. Kiểm tra qua cơ chế FileHandle của libGDX (Internal & Classpath)
        for (String name : candidateNames) {
            try {
                FileHandle fh = Gdx.files.internal(name);
                if (fh.exists() && !fh.isDirectory()) {
                    System.out.println("[AudioManager] Đã tìm thấy nhạc nền qua Gdx.files.internal: " + fh.path());
                    return fh;
                }
            } catch (Exception ignored) {
            }

            try {
                String cleanClasspathPath = name.startsWith("/") ? name.substring(1) : name;
                FileHandle fh = Gdx.files.classpath(cleanClasspathPath);
                if (fh.exists() && !fh.isDirectory()) {
                    System.out.println("[AudioManager] Đã tìm thấy nhạc nền qua Gdx.files.classpath: " + fh.path());
                    return fh;
                }
            } catch (Exception ignored) {
            }

            // 2. Kiểm tra theo các thư mục trên ổ cứng (File hệ thống)
            for (String dir : baseDirs) {
                File f = new File(dir + name);
                if (f.exists() && f.isFile()) {
                    System.out.println("[AudioManager] Đã nạp nhạc nền từ ổ cứng: " + f.getPath());
                    return Gdx.files.absolute(f.getAbsolutePath());
                }
            }
        }

        // 3. Nếu người dùng đặt file .mp3 ở một thư mục con bất kỳ trong project, tự động quét tìm file .mp3 đầu tiên
        try {
            Path rootDir = Path.of(".");
            if (Files.exists(rootDir)) {
                try (Stream<Path> stream = Files.walk(rootDir, 4)) {
                    Path foundMp3 = stream
                            .filter(Files::isRegularFile)
                            .filter(p -> p.getFileName().toString().toLowerCase().endsWith(".mp3"))
                            .findFirst()
                            .orElse(null);

                    if (foundMp3 != null) {
                        System.out.println("[AudioManager] Tự động phát hiện và nạp nhạc nền: " + foundMp3);
                        return Gdx.files.absolute(foundMp3.toAbsolutePath().toString());
                    }
                }
            }
        } catch (IOException ignored) {
        }

        return null;
    }
}

package com.voxelgame.engine;

import com.badlogic.gdx.math.Matrix4;
import com.badlogic.gdx.math.Vector3;

/**
 * Quản lý góc nhìn thứ nhất (First-Person Camera) trong không gian 3 chiều và phép chiếu 2D cho giao diện.
 *
 * THỂ HIỆN TÍNH ĐÓNG GÓI (ENCAPSULATION - DEFENSIVE COPYING):
 * - Toàn bộ các vector cơ sở nội bộ (`position`, `front`, `up`, `right`, `worldUp`) được khóa `private final`.
 * - Các phương thức truy vấn `getPosition()`, `getFront()`, `getRight()`, `getUp()` luôn trả về một bản sao
 *   độc lập (`new Vector3(...)`), ngăn chặn tuyệt đối việc các lớp bên ngoài vô tình gọi sửa trực tiếp
 *   làm biến dạng hệ trục tọa độ của Camera.
 */
public class Camera {

    // Tọa độ vị trí mắt của Camera trong không gian thế giới (World Space)
    private final Vector3 position;

    // 3 vector cơ sở vuông góc từng đôi một xác định hệ trục tọa độ gắn liền với mắt người chơi
    private final Vector3 front;   // Hướng đang nhìn thẳng tới trước
    private final Vector3 up;      // Hướng đỉnh đầu của Camera
    private final Vector3 right;   // Hướng bên tay phải của Camera
    private final Vector3 worldUp; // Hướng thẳng đứng cố định của thế giới (0, 1, 0)

    // Vector đích nhìn tạm thời được tái sử dụng để tránh cấp phát rác GC trong Game Loop
    private final Vector3 lookTarget = new Vector3();

    // Góc quay Euler (đơn vị: độ - Degrees)
    private float yaw = -90.0f;
    private float pitch = 0.0f;

    // Các thông số ống kính 3D (Frustum)
    private static final float BASE_FOV_DEG = 70.0f;
    private float currentFovDeg = BASE_FOV_DEG;
    private static final float NEAR_PLANE = 0.08f;
    private static final float FAR_PLANE = 160.0f;

    // Bộ đệm ma trận được tái sử dụng mỗi khung hình để tránh tạo rác bộ nhớ (Garbage Collection)
    private final Matrix4 viewMatrix = new Matrix4();
    private final Matrix4 projectionMatrix = new Matrix4();
    private final Matrix4 orthoMatrix = new Matrix4();

    public Camera() {
        this.position = new Vector3(0.0f, 0.0f, 0.0f);
        this.front = new Vector3(0.0f, 0.0f, -1.0f);
        this.up = new Vector3(0.0f, 1.0f, 0.0f);
        this.right = new Vector3(1.0f, 0.0f, 0.0f);
        this.worldUp = new Vector3(0.0f, 1.0f, 0.0f);

        updateCameraVectors();
    }

    /**
     * Xoay góc nhìn Camera dựa trên độ dịch chuyển của con trỏ chuột.
     *
     * @param yawOffset độ dịch chuyển góc nhìn ngang
     * @param pitchOffset độ dịch chuyển góc nhìn dọc
     */
    public void rotate(float yawOffset, float pitchOffset) {
        yaw += yawOffset;
        pitch += pitchOffset;

        // Giới hạn góc nhìn dọc trong khoảng [-89.0, 89.0] độ để chống hiện tượng Gimbal Lock
        if (pitch > 89.0f) {
            pitch = 89.0f;
        }
        if (pitch < -89.0f) {
            pitch = -89.0f;
        }

        updateCameraVectors();
    }

    private void updateCameraVectors() {
        double yawRad = Math.toRadians(yaw);
        double pitchRad = Math.toRadians(pitch);

        front.x = (float) (Math.cos(yawRad) * Math.cos(pitchRad));
        front.y = (float) Math.sin(pitchRad);
        front.z = (float) (Math.sin(yawRad) * Math.cos(pitchRad));
        front.nor();

        right.set(front).crs(worldUp).nor();
        up.set(right).crs(front).nor();
    }

    /**
     * Cập nhật hiệu ứng mở rộng nhẹ góc nhìn (Dynamic FOV) khi người chơi Chạy nhanh (Sprint) hoặc Bay (Fly).
     *
     * @param isSprinting cờ người chơi đang chạy nhanh
     * @param isFlying cờ người chơi đang bay
     * @param deltaTime thời gian giữa hai khung hình
     */
    public void updateDynamicFov(boolean isSprinting, boolean isFlying, float deltaTime) {
        float targetFovDeg = BASE_FOV_DEG;
        if (isSprinting) {
            targetFovDeg += 8.0f;
        }
        if (isFlying) {
            targetFovDeg += 5.0f;
        }
        float lerpSpeed = Math.min(1.0f, deltaTime * 10.0f);
        currentFovDeg += (targetFovDeg - currentFovDeg) * lerpSpeed;
    }

    /**
     * Trả về Ma trận Nhìn (View Matrix) dùng phép biến đổi `lookAt(eye, center, up)`.
     * Tái sử dụng đối tượng viewMatrix có sẵn, không cấp phát thêm bộ nhớ.
     *
     * @return ma trận nhìn Matrix4
     */
    public Matrix4 getViewMatrix() {
        lookTarget.set(position).add(front);
        return viewMatrix.setToLookAt(position, lookTarget, up);
    }

    /**
     * Trả về Ma trận Chiếu phối cảnh 3D (Perspective Projection Matrix).
     *
     * @param windowWidth chiều rộng màn hình
     * @param windowHeight chiều cao màn hình
     * @return ma trận chiếu phối cảnh Matrix4
     */
    public Matrix4 getProjectionMatrix(int windowWidth, int windowHeight) {
        float aspectRatio = (float) windowWidth / (float) Math.max(1, windowHeight);
        return projectionMatrix.setToProjection(NEAR_PLANE, FAR_PLANE, currentFovDeg, aspectRatio);
    }

    /**
     * Trả về Ma trận Chiếu trực giao 2D (Orthographic Projection Matrix) phục vụ vẽ giao diện (UI).
     * Trục Y đi từ 0 (đỉnh màn hình) xuống windowHeight (đáy màn hình), khớp 100% với hệ tọa độ Top-Left.
     *
     * @param windowWidth chiều rộng màn hình
     * @param windowHeight chiều cao màn hình
     * @return ma trận trực giao Matrix4
     */
    public Matrix4 getOrthoMatrix(int windowWidth, int windowHeight) {
        return orthoMatrix.setToOrtho(0.0f, windowWidth, windowHeight, 0.0f, -1.0f, 1.0f);
    }

    /**
     * Trả về bản sao (Defensive Copy) vị trí mắt Camera.
     *
     * @return đối tượng Vector3 mới chứa vị trí
     */
    public Vector3 getPosition() {
        return new Vector3(position);
    }

    public void setPosition(float x, float y, float z) {
        this.position.set(x, y, z);
    }

    /**
     * Trả về bản sao (Defensive Copy) vector hướng nhìn phía trước.
     *
     * @return đối tượng Vector3 mới chứa hướng nhìn
     */
    public Vector3 getFront() {
        return new Vector3(front);
    }

    /**
     * Trả về bản sao (Defensive Copy) vector hướng bên phải.
     *
     * @return đối tượng Vector3 mới chứa hướng phải
     */
    public Vector3 getRight() {
        return new Vector3(right);
    }

    /**
     * Trả về bản sao (Defensive Copy) vector hướng đỉnh đầu Camera.
     *
     * @return đối tượng Vector3 mới chứa hướng đỉnh đầu
     */
    public Vector3 getUp() {
        return new Vector3(up);
    }

    public float getYaw() {
        return yaw;
    }

    public float getPitch() {
        return pitch;
    }
}

package com.voxelgame.engine;

import com.badlogic.gdx.graphics.GL20;
import com.badlogic.gdx.graphics.Mesh;
import com.badlogic.gdx.graphics.VertexAttribute;
import com.badlogic.gdx.graphics.VertexAttributes;
import com.badlogic.gdx.graphics.VertexAttributes.Usage;

/**
 * Quản lý một phân vùng không gian 16x16x16 Blocks (Chunk) và bộ đệm lưới đỉnh (libGDX Mesh),
 * triển khai giao diện {@link Disposable}.
 *
 * THỂ HIỆN CÁC TÍNH CHẤT OOP:
 * 1. Tính Đóng gói cấp Package (Package-Private Encapsulation):
 *    - Các phương thức thao tác nội bộ (`getBlockLocal`, `setBlockLocal`, `isDirty`, `markDirty`,
 *      `rebuildMesh`, `render`) được giới hạn ở mức truy cập **package-private** (không có `public`).
 *    - Nhờ đó, chỉ lớp điều phối `World` nằm cùng package `com.voxelgame.engine` mới được phép can thiệp
 *      vào mảng dữ liệu và quá trình tái tạo lưới đỉnh Mesh của Chunk.
 * 2. Tính Đa hình (Polymorphism):
 *    - Trong `rebuildMesh(World)`, Chunk gọi trực tiếp phương thức đa hình `block.getTopHeight(world, wx, wy, wz)`
 *      và `block.getFaceUV(Block.Face.*)` để mỗi loại khối tự quyết định hình học và tọa độ chất liệu.
 */
public class Chunk implements Disposable {

    public static final int SIZE = 16;
    private static final int FLOATS_PER_VERTEX = 6; // [x, y, z, u, v, shade]

    // Định nghĩa cấu trúc thuộc tính đỉnh của libGDX: vị trí vec3 (0), uv vec2 (1), shade float (2)
    private static final VertexAttributes VERTEX_ATTRIBUTES = new VertexAttributes(
            new VertexAttribute(Usage.Position, 3, "aPos"),
            new VertexAttribute(Usage.TextureCoordinates, 2, "aTexCoord"),
            new VertexAttribute(Usage.Generic, 1, "aShade")
    );

    private final int chunkX;
    private final int chunkY;
    private final int chunkZ;

    // Mảng 3D lưu trữ mã ID (1 byte/ô) của các Block trong Chunk
    private final byte[][][] blocks = new byte[SIZE][SIZE][SIZE];

    private Mesh mesh;
    private int vertexCount = 0;
    private boolean dirty = true;

    // Hệ số độ sáng theo hướng mặt (Directional Face Shading)
    private static final float SHADE_TOP = 1.00f;
    private static final float SHADE_BOTTOM = 0.55f;
    private static final float SHADE_NORTH_SOUTH = 0.85f;
    private static final float SHADE_EAST_WEST = 0.72f;

    public Chunk(int chunkX, int chunkY, int chunkZ) {
        this.chunkX = chunkX;
        this.chunkY = chunkY;
        this.chunkZ = chunkZ;
    }

    byte getBlockLocal(int x, int y, int z) {
        if (x < 0 || x >= SIZE || y < 0 || y >= SIZE || z < 0 || z >= SIZE) {
            return 0;
        }
        return blocks[x][y][z];
    }

    void setBlockLocal(int x, int y, int z, byte blockId) {
        if (x >= 0 && x < SIZE && y >= 0 && y < SIZE && z >= 0 && z < SIZE) {
            blocks[x][y][z] = blockId;
            dirty = true;
        }
    }

    boolean isDirty() {
        return dirty;
    }

    void markDirty() {
        this.dirty = true;
    }

    /**
     * Xây dựng lại lưới đỉnh (Mesh Rebuilding) cho toàn bộ Chunk dựa trên đối tượng đa hình {@link Block}.
     * Chỉ cho phép gọi trong nội bộ package `com.voxelgame.engine`.
     *
     * @param world thế giới voxel cung cấp dữ liệu các khối lân cận
     */
    void rebuildMesh(World world) {
        List<Float> vertices = new ArrayList<>(4096);

        int baseWorldX = chunkX * SIZE;
        int baseWorldY = chunkY * SIZE;
        int baseWorldZ = chunkZ * SIZE;

        for (int x = 0; x < SIZE; x++) {
            for (int y = 0; y < SIZE; y++) {
                for (int z = 0; z < SIZE; z++) {
                    byte blockId = blocks[x][y][z];
                    if (blockId == 0) {
                        continue;
                    }

                    Block block = Block.fromId(blockId);
                    if (block == Block.AIR) {
                        continue;
                    }

                    int wx = baseWorldX + x;
                    int wy = baseWorldY + y;
                    int wz = baseWorldZ + z;

                    // Gọi đa hình chiều cao bề mặt khối (1.0f cho khối rắn, 0.88f cho mặt thoáng Nước/Lava)
                    float topHeight = block.getTopHeight(world, wx, wy, wz);

                    // 1. Mặt Trên (+Y)
                    if (shouldRenderFace(block, Block.fromId(world.getBlock(wx, wy + 1, wz)))) {
                        addTopFace(vertices, wx, wy, wz, topHeight, block.getFaceUV(Block.Face.TOP));
                    }

                    // 2. Mặt Dưới (-Y)
                    if (shouldRenderFace(block, Block.fromId(world.getBlock(wx, wy - 1, wz)))) {
                        addBottomFace(vertices, wx, wy, wz, block.getFaceUV(Block.Face.BOTTOM));
                    }

                    // 3. Mặt Trước / Nam (+Z)
                    if (shouldRenderFace(block, Block.fromId(world.getBlock(wx, wy, wz + 1)))) {
                        addFrontFace(vertices, wx, wy, wz, topHeight, block.getFaceUV(Block.Face.SOUTH));
                    }

                    // 4. Mặt Sau / Bắc (-Z)
                    if (shouldRenderFace(block, Block.fromId(world.getBlock(wx, wy, wz - 1)))) {
                        addBackFace(vertices, wx, wy, wz, topHeight, block.getFaceUV(Block.Face.NORTH));
                    }

                    // 5. Mặt Phải / Đông (+X)
                    if (shouldRenderFace(block, Block.fromId(world.getBlock(wx + 1, wy, wz)))) {
                        addRightFace(vertices, wx, wy, wz, topHeight, block.getFaceUV(Block.Face.EAST));
                    }

                    // 6. Mặt Trái / Tây (-X)
                    if (shouldRenderFace(block, Block.fromId(world.getBlock(wx - 1, wy, wz)))) {
                        addLeftFace(vertices, wx, wy, wz, topHeight, block.getFaceUV(Block.Face.WEST));
                    }
                }
            }
        }

        this.vertexCount = vertices.size() / FLOATS_PER_VERTEX;

        if (vertexCount > 0) {
            float[] vertexArray = new float[vertices.size()];
            for (int i = 0; i < vertices.size(); i++) {
                vertexArray[i] = vertices.get(i);
            }

            if (mesh == null || mesh.getMaxVertices() < vertexCount) {
                if (mesh != null) {
                    mesh.dispose();
                }
                mesh = new Mesh(true, vertexCount, 0, VERTEX_ATTRIBUTES);
            }
            mesh.setVertices(vertexArray, 0, vertexArray.length);
        } else {
            if (mesh != null) {
                mesh.dispose();
                mesh = null;
            }
        }

        this.dirty = false;
    }

    private boolean shouldRenderFace(Block current, Block neighbor) {
        if (current.isLiquid()) {
            return neighbor == Block.AIR;
        }
        return !neighbor.isSolid();
    }

    void render(ShaderProgram shader) {
        if (mesh != null && vertexCount > 0) {
            mesh.render(shader.getGdxShader(), GL20.GL_TRIANGLES, 0, vertexCount);
        }
    }

    @Override
    public void dispose() {
        if (mesh != null) {
            mesh.dispose();
            mesh = null;
        }
    }

    private void putVertex(List<Float> list, float x, float y, float z, float u, float v, float shade) {
        list.add(x);
        list.add(y);
        list.add(z);
        list.add(u);
        list.add(v);
        list.add(shade);
    }

    private void addTopFace(List<Float> v, float x, float y, float z, float h, float[] uv) {
        float u0 = uv[0], v0 = uv[1], u1 = uv[2], v1 = uv[3];
        putVertex(v, x,     y + h, z + 1, u0, v1, SHADE_TOP);
        putVertex(v, x + 1, y + h, z + 1, u1, v1, SHADE_TOP);
        putVertex(v, x + 1, y + h, z,     u1, v0, SHADE_TOP);

        putVertex(v, x,     y + h, z + 1, u0, v1, SHADE_TOP);
        putVertex(v, x + 1, y + h, z,     u1, v0, SHADE_TOP);
        putVertex(v, x,     y + h, z,     u0, v0, SHADE_TOP);
    }

    private void addBottomFace(List<Float> v, float x, float y, float z, float[] uv) {
        float u0 = uv[0], v0 = uv[1], u1 = uv[2], v1 = uv[3];
        putVertex(v, x,     y, z,     u0, v1, SHADE_BOTTOM);
        putVertex(v, x + 1, y, z,     u1, v1, SHADE_BOTTOM);
        putVertex(v, x + 1, y, z + 1, u1, v0, SHADE_BOTTOM);

        putVertex(v, x,     y, z,     u0, v1, SHADE_BOTTOM);
        putVertex(v, x + 1, y, z + 1, u1, v0, SHADE_BOTTOM);
        putVertex(v, x,     y, z + 1, u0, v0, SHADE_BOTTOM);
    }

    private void addFrontFace(List<Float> v, float x, float y, float z, float h, float[] uv) {
        float u0 = uv[0], v0 = uv[1], u1 = uv[2], v1 = uv[3];
        putVertex(v, x,     y,     z + 1, u0, v1, SHADE_NORTH_SOUTH);
        putVertex(v, x + 1, y,     z + 1, u1, v1, SHADE_NORTH_SOUTH);
        putVertex(v, x + 1, y + h, z + 1, u1, v0, SHADE_NORTH_SOUTH);

        putVertex(v, x,     y,     z + 1, u0, v1, SHADE_NORTH_SOUTH);
        putVertex(v, x + 1, y + h, z + 1, u1, v0, SHADE_NORTH_SOUTH);
        putVertex(v, x,     y + h, z + 1, u0, v0, SHADE_NORTH_SOUTH);
    }

    private void addBackFace(List<Float> v, float x, float y, float z, float h, float[] uv) {
        float u0 = uv[0], v0 = uv[1], u1 = uv[2], v1 = uv[3];
        putVertex(v, x + 1, y,     z, u0, v1, SHADE_NORTH_SOUTH);
        putVertex(v, x,     y,     z, u1, v1, SHADE_NORTH_SOUTH);
        putVertex(v, x,     y + h, z, u1, v0, SHADE_NORTH_SOUTH);

        putVertex(v, x + 1, y,     z, u0, v1, SHADE_NORTH_SOUTH);
        putVertex(v, x,     y + h, z, u1, v0, SHADE_NORTH_SOUTH);
        putVertex(v, x + 1, y + h, z, u0, v0, SHADE_NORTH_SOUTH);
    }

    private void addRightFace(List<Float> v, float x, float y, float z, float h, float[] uv) {
        float u0 = uv[0], v0 = uv[1], u1 = uv[2], v1 = uv[3];
        putVertex(v, x + 1, y,     z + 1, u0, v1, SHADE_EAST_WEST);
        putVertex(v, x + 1, y,     z,     u1, v1, SHADE_EAST_WEST);
        putVertex(v, x + 1, y + h, z,     u1, v0, SHADE_EAST_WEST);

        putVertex(v, x + 1, y,     z + 1, u0, v1, SHADE_EAST_WEST);
        putVertex(v, x + 1, y + h, z,     u1, v0, SHADE_EAST_WEST);
        putVertex(v, x + 1, y + h, z + 1, u0, v0, SHADE_EAST_WEST);
    }

    private void addLeftFace(List<Float> v, float x, float y, float z, float h, float[] uv) {
        float u0 = uv[0], v0 = uv[1], u1 = uv[2], v1 = uv[3];
        putVertex(v, x, y,     z,     u0, v1, SHADE_EAST_WEST);
        putVertex(v, x, y,     z + 1, u1, v1, SHADE_EAST_WEST);
        putVertex(v, x, y + h, z + 1, u1, v0, SHADE_EAST_WEST);

        putVertex(v, x, y,     z,     u0, v1, SHADE_EAST_WEST);
        putVertex(v, x, y + h, z + 1, u1, v0, SHADE_EAST_WEST);
        putVertex(v, x, y + h, z,     u0, v0, SHADE_EAST_WEST);
    }
}

package com.voxelgame.engine;

import com.badlogic.gdx.math.Vector3;
import com.voxelgame.entity.Block;

/**
 * Thực hiện thuật toán bắn tia 3D DDA (Digital Differential Analyzer -
 * Amanatides & Woo)
 * trong không gian lưới Voxel 3D.
 *
 * THỂ HIỆN TÍNH ĐÓNG GÓI (ENCAPSULATION) & ĐA HÌNH (POLYMORPHISM):
 * - Lớp kết quả `HitResult` đóng gói toàn bộ các trường tọa độ ở phạm vi
 * `private final` và chỉ
 * cung cấp các phương thức truy vấn chuẩn (`isHit()`, `getBlockX()`,
 * `getBlockY()`, `getBlockZ()`,
 * `getPrevBlockX()`, `getPrevBlockY()`, `getPrevBlockZ()`, `getBlock()`).
 * - Tia Raycast kiểm tra tính chất vật lý của mỗi ô thông qua đối tượng đa hình
 * `Block.fromId(blockId)`:
 * tự động đi xuyên qua Không khí (`AirBlock`) và Chất lỏng (`WaterBlock`,
 * `LavaBlock`), chỉ dừng lại
 * khi chạm vào một khối rắn (`block.isSolid() == true`).
 */
public class Raycast {

    /**
     * Cấu trúc dữ liệu bất biến (Immutable Value Object) lưu trữ kết quả bắn tia
     * Raycast.
     */
    public static class HitResult {
        private final boolean hit;
        private final int blockX;
        private final int blockY;
        private final int blockZ;
        private final int prevBlockX;
        private final int prevBlockY;
        private final int prevBlockZ;
        private final Block block;

        public HitResult(
                boolean hit,
                int blockX,
                int blockY,
                int blockZ,
                int prevBlockX,
                int prevBlockY,
                int prevBlockZ,
                Block block) {
            this.hit = hit;
            this.blockX = blockX;
            this.blockY = blockY;
            this.blockZ = blockZ;
            this.prevBlockX = prevBlockX;
            this.prevBlockY = prevBlockY;
            this.prevBlockZ = prevBlockZ;
            this.block = (block != null) ? block : Block.AIR;
        }

        public static HitResult miss() {
            return new HitResult(false, 0, 0, 0, 0, 0, 0, Block.AIR);
        }

        public boolean isHit() {
            return hit;
        }

        public int getBlockX() {
            return blockX;
        }

        public int getBlockY() {
            return blockY;
        }

        public int getBlockZ() {
            return blockZ;
        }

        public int getPrevBlockX() {
            return prevBlockX;
        }

        public int getPrevBlockY() {
            return prevBlockY;
        }

        public int getPrevBlockZ() {
            return prevBlockZ;
        }

        public Block getBlock() {
            return block;
        }

        public Block getBlockType() {
            return block;
        }
    }

    /**
     * Bắn một tia từ vị trí mắt Camera (`origin`) theo hướng nhìn (`direction`) với
     * tầm với tối đa (`maxDistance`).
     * Tự động bỏ qua Không khí (`Block.AIR`) và các khối Chất lỏng (`WaterBlock`,
     * `LavaBlock`).
     *
     * @param world       thế giới voxel
     * @param origin      vị trí điểm gốc của tia (mắt người chơi)
     * @param direction   hướng nhìn 3D (vector chỉ phương)
     * @param maxDistance tầm với xa nhất của tia
     * @return đối tượng HitResult chứa thông tin khối bị chạm hoặc HitResult.miss()
     */
    public static HitResult cast(World world, Vector3 origin, Vector3 direction, float maxDistance) {
        Vector3 dir = new Vector3(direction);
        if (dir.len2() < 1e-8f) {
            return HitResult.miss();
        }
        dir.nor();

        int mapX = (int) Math.floor(origin.x);
        int mapY = (int) Math.floor(origin.y);
        int mapZ = (int) Math.floor(origin.z);

        int prevX = mapX;
        int prevY = mapY;
        int prevZ = mapZ;

        int stepX = dir.x >= 0 ? 1 : -1;
        int stepY = dir.y >= 0 ? 1 : -1;
        int stepZ = dir.z >= 0 ? 1 : -1;

        float tDeltaX = (Math.abs(dir.x) < 1e-8f) ? Float.POSITIVE_INFINITY : Math.abs(1.0f / dir.x);
        float tDeltaY = (Math.abs(dir.y) < 1e-8f) ? Float.POSITIVE_INFINITY : Math.abs(1.0f / dir.y);
        float tDeltaZ = (Math.abs(dir.z) < 1e-8f) ? Float.POSITIVE_INFINITY : Math.abs(1.0f / dir.z);

        float nextVoxelBoundaryX = (stepX > 0) ? (mapX + 1.0f) : mapX;
        float nextVoxelBoundaryY = (stepY > 0) ? (mapY + 1.0f) : mapY;
        float nextVoxelBoundaryZ = (stepZ > 0) ? (mapZ + 1.0f) : mapZ;

        float tMaxX = (Math.abs(dir.x) < 1e-8f) ? Float.POSITIVE_INFINITY : (nextVoxelBoundaryX - origin.x) / dir.x;
        float tMaxY = (Math.abs(dir.y) < 1e-8f) ? Float.POSITIVE_INFINITY : (nextVoxelBoundaryY - origin.y) / dir.y;
        float tMaxZ = (Math.abs(dir.z) < 1e-8f) ? Float.POSITIVE_INFINITY : (nextVoxelBoundaryZ - origin.z) / dir.z;

        float traveledDistance = 0.0f;

        while (traveledDistance <= maxDistance) {
            byte blockId = world.getBlock(mapX, mapY, mapZ);
            Block block = Block.fromId(blockId);

            // Chỉ dừng lại khi gặp một Block RẮN (block.isSolid() == true)
            if (block.isSolid()) {
                return new HitResult(true, mapX, mapY, mapZ, prevX, prevY, prevZ, block);
            }

            prevX = mapX;
            prevY = mapY;
            prevZ = mapZ;

            if (tMaxX < tMaxY) {
                if (tMaxX < tMaxZ) {
                    mapX += stepX;
                    traveledDistance = tMaxX;
                    tMaxX += tDeltaX;
                } else {
                    mapZ += stepZ;
                    traveledDistance = tMaxZ;
                    tMaxZ += tDeltaZ;
                }
            } else {
                if (tMaxY < tMaxZ) {
                    mapY += stepY;
                    traveledDistance = tMaxY;
                    tMaxY += tDeltaY;
                } else {
                    mapZ += stepZ;
                    traveledDistance = tMaxZ;
                    tMaxZ += tDeltaZ;
                }
            }
        }

        return HitResult.miss();
    }
}

package com.voxelgame.engine;

import com.badlogic.gdx.Gdx;
import com.badlogic.gdx.math.Matrix4;
import com.badlogic.gdx.math.Vector3;
import com.badlogic.gdx.utils.Disposable;

/**
 * Quản lý việc biên dịch (Compile), liên kết (Link) chương trình GLSL Shader 330 Core trên GPU
 * và truyền các biến Uniform từ Java xuống GPU thông qua libGDX, triển khai {@link Disposable}.
 *
 * THỂ HIỆN CÁC TÍNH CHẤT OOP:
 * 1. Tính Trừu tượng (Abstraction):
 *    - Ẩn toàn bộ thao tác GLSL cấp thấp đằng sau các phương thức trực quan như
 *      `enable3DWorldMode()`, `enable2DUIMode()`, `setUniformMatrix4f()`.
 * 2. Tính Đa hình lúc biên dịch (Compile-time Polymorphism / Method Overloading):
 *    - Nạp chồng phương thức `setUniformVec4(String, float, float, float, float)` và
 *      `setUniformVec3(String, Vector3)` giúp truyền màu sắc hoặc vector linh hoạt.
 * 3. Tính Đóng gói (Encapsulation):
 *    - Đóng gói đối tượng `com.badlogic.gdx.graphics.glutils.ShaderProgram` ở phạm vi `private final`.
 */
public class ShaderProgram implements Disposable {

    private final com.badlogic.gdx.graphics.glutils.ShaderProgram gdxShader;
    private final Matrix4 identityMatrix = new Matrix4();

    /**
     * Mã nguồn Vertex Shader (GLSL 330 Core).
     */
    private static final String VERTEX_SHADER_SOURCE = """
            #version 330 core
            layout (location = 0) in vec3 aPos;
            layout (location = 1) in vec2 aTexCoord;
            layout (location = 2) in float aShade;

            out vec2 vTexCoord;
            out float vShade;
            out float vFogDist;

            uniform mat4 uModel;
            uniform mat4 uView;
            uniform mat4 uProjection;

            void main() {
                vec4 worldPos = uModel * vec4(aPos, 1.0);
                vec4 viewPos = uView * worldPos;

                gl_Position = uProjection * viewPos;

                vTexCoord = aTexCoord;
                vShade = aShade;
                // Khoảng cách Euclid từ mắt Camera (gốc tọa độ trong View Space) tới điểm hiện tại
                vFogDist = length(viewPos.xyz);
            }
            """;

    /**
     * Mã nguồn Fragment Shader (GLSL 330 Core).
     */
    private static final String FRAGMENT_SHADER_SOURCE = """
            #version 330 core
            in vec2 vTexCoord;
            in float vShade;
            in float vFogDist;

            out vec4 FragColor;

            uniform sampler2D uTexture;
            uniform int uUseTexture;
            uniform int uUseFog;
            uniform vec4 uTintColor;

            uniform vec3 uFogColor;
            uniform float uFogStart;
            uniform float uFogEnd;

            void main() {
                // 1. Xác định màu cơ sở (lấy từ Texture Atlas hoặc dùng màu thuần túy cho 2D UI)
                vec4 baseColor;
                if (uUseTexture == 1) {
                    baseColor = texture(uTexture, vTexCoord) * uTintColor;
                } else {
                    baseColor = uTintColor;
                }

                // Loại bỏ các điểm ảnh hoàn toàn trong suốt (Alpha Testing cơ bản)
                if (baseColor.a < 0.05) {
                    discard;
                }

                // 2. Nhân với hệ số độ sáng theo hướng mặt khối (Ambient/Directional Face Shading)
                vec3 litColor = baseColor.rgb * vShade;

                // 3. Nếu đang bật sương mù (khi vẽ 3D World), hòa trộn màu khối với màu bầu trời ở xa
                vec3 finalRgb = litColor;
                if (uUseFog == 1) {
                    float denom = max(0.0001, uFogEnd - uFogStart);
                    float fogFactor = clamp((uFogEnd - vFogDist) / denom, 0.0, 1.0);
                    finalRgb = mix(uFogColor, litColor, fogFactor);
                }

                FragColor = vec4(finalRgb, baseColor.a);
            }
            """;

    public ShaderProgram() {
        // Tắt chuỗi tiền tố mặc định của libGDX để giữ nguyên #version 330 core ở dòng 1
        com.badlogic.gdx.graphics.glutils.ShaderProgram.prependVertexCode = "";
        com.badlogic.gdx.graphics.glutils.ShaderProgram.prependFragmentCode = "";

        gdxShader = new com.badlogic.gdx.graphics.glutils.ShaderProgram(
                VERTEX_SHADER_SOURCE,
                FRAGMENT_SHADER_SOURCE
        );

        if (!gdxShader.isCompiled()) {
            throw new RuntimeException("Lỗi biên dịch GLSL Shader:\n" + gdxShader.getLog());
        }

        bind();
        setUniform1i("uTexture", 0);
        setUniform1i("uUseTexture", 1);
        setUniform1i("uUseFog", 1);

    private void blitResizedToTile(
            Pixmap atlasPixmap,
            Pixmap src,
            int srcX,
            int srcY,
            int srcW,
            int srcH,
            int tileIndex,
            boolean tintWaterBlue
    ) {
        for (int ty = 0; ty < TILE_SIZE; ty++) {
            for (int tx = 0; tx < TILE_SIZE; tx++) {
                int sx = srcX + Math.min(srcW - 1, (int) (((tx + 0.5f) / TILE_SIZE) * srcW));
                int sy = srcY + Math.min(srcH - 1, (int) (((ty + 0.5f) / TILE_SIZE) * srcH));

                int pixel = src.getPixel(sx, sy);
                int r = (pixel >>> 24) & 0xFF;
                int g = (pixel >>> 16) & 0xFF;
                int b = (pixel >>> 8) & 0xFF;
                int a = pixel & 0xFF;

                if (tintWaterBlue && Math.abs(r - g) < 10 && Math.abs(g - b) < 10) {
                    r = clamp((int) (r * 0.22f + 18));
                    g = clamp((int) (g * 0.58f + 45));
                    b = clamp((int) (b * 0.98f + 70));
                    a = 225;
                } else {
                    a = 255;
                }

                putAtlasPixel(atlasPixmap, tileIndex, tx, ty, r, g, b, a);
            }
        }
    }

    private void putAtlasPixel(Pixmap atlasPixmap, int tileIndex, int tx, int ty, int r, int g, int b, int a) {
        int tileCol = tileIndex % ATLAS_COLS;
        int tileRow = tileIndex / ATLAS_COLS;

        int atlasX = tileCol * TILE_SIZE + tx;
        int atlasY = tileRow * TILE_SIZE + ty;

        int rgba = (clamp(r) << 24) | (clamp(g) << 16) | (clamp(b) << 8) | clamp(a);
        atlasPixmap.drawPixel(atlasX, atlasY, rgba);
    }

        int bx = hit.getBlockX();
        int by = hit.getBlockY();
        int bz = hit.getBlockZ();

        // Phóng to nhẹ khối bọc ngoài thêm 0.004f mỗi bên để không bị trùng mặt phẳng (Z-Fighting) với Block gốc
        float expand = 0.004f;
        float x0 = bx - expand;
        float y0 = by - expand;
        float z0 = bz - expand;
        float x1 = bx + 1.0f + expand;
        float y1 = by + 1.0f + expand;
        float z1 = bz + 1.0f + expand;

        modelMatrix.idt();
        shader.setUniformMatrix4f("uModel", modelMatrix);

        if (miningProgress > 0.001f) {
            // Khi đang giữ chuột trái đập block: Vẽ lớp phủ nứt/tối dần và co giãn theo phần trăm tiến trình
            float alpha = Math.min(0.68f, 0.18f + miningProgress * 0.50f);
            shader.setUniformVec4("uTintColor", 0.08f, 0.08f, 0.08f, alpha);
            draw3DBoxSolid(x0, y0, z0, x1, y1, z1);

            // Vẽ thêm lõi chỉ báo vết nứt co dần ở tâm mỗi mặt
            float crackRatio = Math.min(1.0f, miningProgress);
            float pad = (1.0f - crackRatio) * 0.45f;
            shader.setUniformVec4("uTintColor", 1.0f, 0.92f, 0.25f, 0.35f);
            draw3DBoxSolid(
                    bx + pad - expand * 2,
                    by + pad - expand * 2,
                    bz + pad - expand * 2,
                    bx + 1.0f - pad + expand * 2,
                    by + 1.0f - pad + expand * 2,
                    bz + 1.0f - pad + expand * 2
            );
        } else {
            // Khi chỉ đang nhìn vào block: phủ lớp sáng mờ cực nhẹ để người chơi biết đang chọn trúng ô nào
            shader.setUniformVec4("uTintColor", 1.0f, 1.0f, 1.0f, 0.09f);
            draw3DBoxSolid(x0, y0, z0, x1, y1, z1);
        }

        shader.setUniform1i("uUseTexture", 1);
        shader.setUniformVec4("uTintColor", 1.0f, 1.0f, 1.0f, 1.0f);

        Gdx.gl.glEnable(GL20.GL_CULL_FACE);
        Gdx.gl.glDisable(GL20.GL_BLEND);
    }

    /**
     * Phương thức nạp chồng tiện ích nhận trực tiếp đối tượng {@link Player} và lấy `player.getHotbar()`.

    public void render2DHUD(

            int windowHeight,
            float miningProgress,
            boolean isPaused,
            double mouseX,

        Hotbar hotbar = (player != null) ? player.getHotbar() : null;
        render2DHUD(
                shader,

                hotbar,
                audioManager,
                windowWidth,
                windowHeight,
                miningProgress,
                isPaused,
                mouseX,

    }

    /**
     * Kết xuất toàn bộ giao diện 2D HUD và Pause Menu lên màn hình bằng phép chiếu Trực giao (Orthographic).
     */
    public void render2DHUD(
            ShaderProgram shader,
            Texture textureAtlas,
            Camera camera,
            Hotbar hotbar,
            AudioManager audioManager,
            int windowWidth,
            int windowHeight,
            float miningProgress,
            boolean isPaused,
            double mouseX,
            double mouseY
    ) {
        this.currentShader = shader;

        Gdx.gl.glDisable(GL20.GL_DEPTH_TEST);
        Gdx.gl.glDisable(GL20.GL_CULL_FACE);
        Gdx.gl.glEnable(GL20.GL_BLEND);
        Gdx.gl.glBlendFunc(GL20.GL_SRC_ALPHA, GL20.GL_ONE_MINUS_SRC_ALPHA);

        shader.enable2DUIMode(camera.getOrthoMatrix(windowWidth, windowHeight));

        // 1. Luôn vẽ thanh 9 ô Hotbar ở cạnh dưới màn hình
        if (hotbar != null) {
            renderHotbar(shader, textureAtlas, hotbar, windowWidth, windowHeight);
        }

        if (!isPaused) {
            // 2. Khi đang chơi: Vẽ tâm ngắm Crosshair ở chính giữa màn hình
            renderCrosshair(shader, windowWidth, windowHeight);

            // 3. Nếu đang giữ chuột trái đập một Block: Vẽ thanh Tiến trình Đập Block ngay dưới tâm ngắm
            if (miningProgress > 0.001f) {
                renderMiningProgressBar(shader, windowWidth, windowHeight, miningProgress);
            }
        } else {
            // 4. Khi bấm ESC tạm dừng: Vẽ Menu Tạm dừng (Pause Menu) với ô Resume và điều chỉnh Âm lượng
            renderPauseMenu(shader, audioManager, windowWidth, windowHeight, mouseX, mouseY);
        }

        shader.enable3DWorldMode();
        Gdx.gl.glDisable(GL20.GL_BLEND);
        Gdx.gl.glEnable(GL20.GL_CULL_FACE);
        Gdx.gl.glEnable(GL20.GL_DEPTH_TEST);
    }

    private void renderCrosshair(ShaderProgram shader, int windowWidth, int windowHeight) {
        float cx = windowWidth * 0.5f;
        float cy = windowHeight * 0.5f;

        shader.setUniform1i("uUseTexture", 0);

        shader.setUniformVec4("uTintColor", 0.0f, 0.0f, 0.0f, 0.65f);
        drawRect2D(cx - 10.0f, cy - 2.0f, 20.0f, 4.0f);
        drawRect2D(cx - 2.0f, cy - 10.0f, 4.0f, 20.0f);

        shader.setUniformVec4("uTintColor", 1.0f, 1.0f, 1.0f, 0.92f);
        drawRect2D(cx - 8.0f, cy - 1.0f, 16.0f, 2.0f);
        drawRect2D(cx - 1.0f, cy - 8.0f, 2.0f, 16.0f);
    }

    private void renderMiningProgressBar(ShaderProgram shader, int windowWidth, int windowHeight, float progress) {
        float barW = 110.0f;
        float barH = 10.0f;
        float bx = (windowWidth - barW) * 0.5f;
        float by = windowHeight * 0.5f + 20.0f;

        shader.setUniform1i("uUseTexture", 0);

        shader.setUniformVec4("uTintColor", 0.08f, 0.08f, 0.10f, 0.85f);
        drawRect2D(bx - 2.0f, by - 2.0f, barW + 4.0f, barH + 4.0f);

        float clamped = Math.max(0.0f, Math.min(1.0f, progress));
        float r = 1.0f - clamped * 0.65f;
        float g = 0.72f + clamped * 0.25f;
        shader.setUniformVec4("uTintColor", r, g, 0.18f, 0.95f);
        drawRect2D(bx, by, barW * clamped, barH);
    }

    /**
     * Vẽ giao diện 9 ô Thanh Công Cụ Nhanh (Hotbar) ở cạnh dưới màn hình sử dụng lớp {@link Block}.
     */
    private void renderHotbar(
            ShaderProgram shader,
            Texture textureAtlas,
            Hotbar hotbar,
            int windowWidth,
            int windowHeight
    ) {
        float totalWidth = Hotbar.SLOT_COUNT * HOTBAR_SLOT_SIZE + (Hotbar.SLOT_COUNT + 1) * HOTBAR_SLOT_PADDING;
        float barHeight = HOTBAR_SLOT_SIZE + HOTBAR_SLOT_PADDING * 2.0f;
        float startX = (windowWidth - totalWidth) * 0.5f;
        float startY = windowHeight - barHeight - 16.0f;

        shader.setUniform1i("uUseTexture", 0);

        // 1. Vẽ tấm nền tổng của thanh Hotbar
        shader.setUniformVec4("uTintColor", 0.10f, 0.11f, 0.14f, 0.82f);
        drawRect2D(startX - 4.0f, startY - 4.0f, totalWidth + 8.0f, barHeight + 8.0f);

        // 2. Vẽ từng ô từ 0 đến 8 (tương ứng phím 1 đến 9)
        for (int i = 0; i < Hotbar.SLOT_COUNT; i++) {
            float slotX = startX + HOTBAR_SLOT_PADDING + i * (HOTBAR_SLOT_SIZE + HOTBAR_SLOT_PADDING);
            float slotY = startY + HOTBAR_SLOT_PADDING;

            boolean isSelected = (i == hotbar.getSelectedSlot());

            if (isSelected) {
                shader.setUniform1i("uUseTexture", 0);
                shader.setUniformVec4("uTintColor", 1.0f, 0.92f, 0.32f, 1.0f);
                drawRect2D(slotX - 3.0f, slotY - 3.0f, HOTBAR_SLOT_SIZE + 6.0f, HOTBAR_SLOT_SIZE + 6.0f);
            }

            shader.setUniform1i("uUseTexture", 0);
            if (isSelected) {
                shader.setUniformVec4("uTintColor", 0.28f, 0.30f, 0.35f, 0.95f);
            } else {
                shader.setUniformVec4("uTintColor", 0.18f, 0.19f, 0.23f, 0.90f);
            }
            drawRect2D(slotX, slotY, HOTBAR_SLOT_SIZE, HOTBAR_SLOT_SIZE);

            // Số thứ tự phím (1..9) nhỏ ở góc trái-trên của ô
            shader.setUniformVec4("uTintColor", 0.72f, 0.74f, 0.78f, 0.85f);
            drawPixelText(String.valueOf(i + 1), slotX + 4.0f, slotY + 4.0f, 1.5f);

            Block itemBlock = hotbar.getSlotBlock(i);
            int count = hotbar.getSlotCount(i);

            if (itemBlock != Block.AIR && count > 0) {
                float iconSize = 28.0f;
                float iconX = slotX + (HOTBAR_SLOT_SIZE - iconSize) * 0.5f;
                float iconY = slotY + (HOTBAR_SLOT_SIZE - iconSize) * 0.5f;

                float[] uv = itemBlock.getFaceUV(Block.Face.SOUTH);
                textureAtlas.bind();
                shader.setUniform1i("uUseTexture", 1);
                shader.setUniformVec4("uTintColor", 1.0f, 1.0f, 1.0f, 1.0f);
                drawTexturedRect2D(iconX, iconY, iconSize, iconSize, uv[0], uv[1], uv[2], uv[3]);
                textureAtlas.unbind();

                // Vẽ số lượng item hiện có ở góc phải-dưới của ô
                shader.setUniform1i("uUseTexture", 0);
                String countStr = String.valueOf(count);
                float textScale = 2.0f;
                float textW = countStr.length() * 6.0f * textScale;
                float tx = slotX + HOTBAR_SLOT_SIZE - textW - 3.0f;
                float ty = slotY + HOTBAR_SLOT_SIZE - 7.0f * textScale - 3.0f;

                shader.setUniformVec4("uTintColor", 0.05f, 0.05f, 0.05f, 0.95f);
                drawPixelText(countStr, tx + 1.5f, ty + 1.5f, textScale);

                shader.setUniformVec4("uTintColor", 1.0f, 1.0f, 1.0f, 1.0f);
                drawPixelText(countStr, tx, ty, textScale);
            }
        }

        // Hiển thị tên loại Block đang cầm ngay phía trên thanh Hotbar
        Block selectedBlock = hotbar.getSelectedBlock();
        if (selectedBlock != Block.AIR && hotbar.getSelectedCount() > 0) {
            String label = selectedBlock.getDisplayName().toUpperCase();
            float scale = 2.0f;
            float labelW = label.length() * 6.0f * scale;
            float lx = (windowWidth - labelW) * 0.5f;
            float ly = startY - 24.0f;

            shader.setUniform1i("uUseTexture", 0);
            shader.setUniformVec4("uTintColor", 0.0f, 0.0f, 0.0f, 0.75f);
            drawPixelText(label, lx + 2.0f, ly + 2.0f, scale);
            shader.setUniformVec4("uTintColor", 1.0f, 0.96f, 0.72f, 1.0f);
            drawPixelText(label, lx, ly, scale);
        }
    }

    // =========================================================================
    // 3. VẼ MENU TẠM DỪNG (PAUSE MENU: NÚT RESUME & THANH ÂM LƯỢNG)
    // =========================================================================

    private void renderPauseMenu(
            ShaderProgram shader,
            AudioManager audioManager,
            int windowWidth,
            int windowHeight,
            double mouseX,
            double mouseY
    ) {
        shader.setUniform1i("uUseTexture", 0);

        // 1. Lớp màn tối bán trong suốt phủ toàn màn hình (Dim Backdrop)
        shader.setUniformVec4("uTintColor", 0.04f, 0.05f, 0.08f, 0.68f);
        drawRect2D(0.0f, 0.0f, windowWidth, windowHeight);

        // 2. Khung hộp thoại trung tâm (Pause Panel)
        float panelX = (windowWidth - PANEL_WIDTH) * 0.5f;
        float panelY = (windowHeight - PANEL_HEIGHT) * 0.5f;

        shader.setUniformVec4("uTintColor", 0.65f, 0.68f, 0.75f, 0.95f);
        drawRect2D(panelX - 3.0f, panelY - 3.0f, PANEL_WIDTH + 6.0f, PANEL_HEIGHT + 6.0f);

        shader.setUniformVec4("uTintColor", 0.14f, 0.16f, 0.21f, 0.96f);
        drawRect2D(panelX, panelY, PANEL_WIDTH, PANEL_HEIGHT);

        // 3. Tiêu đề "GAME PAUSED"
        String title = "GAME PAUSED";
        float titleScale = 3.2f;
        float titleW = title.length() * 6.0f * titleScale;
        float titleX = (windowWidth - titleW) * 0.5f;
        float titleY = panelY + 24.0f;

        shader.setUniformVec4("uTintColor", 0.05f, 0.05f, 0.05f, 0.90f);
        drawPixelText(title, titleX + 2.5f, titleY + 2.5f, titleScale);
        shader.setUniformVec4("uTintColor", 1.0f, 0.92f, 0.35f, 1.0f);
        drawPixelText(title, titleX, titleY, titleScale);

        // 4. Ô Nút bấm "RESUME GAME"
        float[] resumeBounds = getResumeButtonBounds(windowWidth, windowHeight);
        boolean hoverResume = isPointInRect(mouseX, mouseY, resumeBounds);

        shader.setUniformVec4("uTintColor", hoverResume ? 1.0f : 0.75f, hoverResume ? 0.95f : 0.78f, 0.40f, 1.0f);
        drawRect2D(resumeBounds[0] - 2.0f, resumeBounds[1] - 2.0f, resumeBounds[2] + 4.0f, resumeBounds[3] + 4.0f);

        if (hoverResume) {
            shader.setUniformVec4("uTintColor", 0.26f, 0.58f, 0.30f, 1.0f);
        } else {
            shader.setUniformVec4("uTintColor", 0.20f, 0.44f, 0.24f, 1.0f);
        }
        drawRect2D(resumeBounds[0], resumeBounds[1], resumeBounds[2], resumeBounds[3]);

        String resumeLabel = "RESUME GAME";
        float resScale = 2.4f;
        float resW = resumeLabel.length() * 6.0f * resScale;
        shader.setUniformVec4("uTintColor", 1.0f, 1.0f, 1.0f, 1.0f);
        drawPixelText(
                resumeLabel,
                resumeBounds[0] + (resumeBounds[2] - resW) * 0.5f,
                resumeBounds[1] + (resumeBounds[3] - 7.0f * resScale) * 0.5f,
                resScale
        );

        // 5. Khu vực Điều chỉnh Âm lượng Nhạc nền (Volume Control: Nút [-], Thanh trượt, Nút [+])
        int volPercent = (audioManager != null) ? audioManager.getVolumePercent() : 70;
        String volLabel = "MUSIC VOLUME: " + volPercent + "%";
        float volLabelScale = 2.0f;
        float volLabelW = volLabel.length() * 6.0f * volLabelScale;
        shader.setUniformVec4("uTintColor", 0.88f, 0.90f, 0.95f, 1.0f);
        drawPixelText(volLabel, (windowWidth - volLabelW) * 0.5f, panelY + 138.0f, volLabelScale);

        // Nút Giảm âm lượng [-]
        float[] minusBounds = getVolumeMinusBounds(windowWidth, windowHeight);
        boolean hoverMinus = isPointInRect(mouseX, mouseY, minusBounds);
        shader.setUniformVec4("uTintColor", 0.75f, 0.78f, 0.82f, 1.0f);
        drawRect2D(minusBounds[0] - 2.0f, minusBounds[1] - 2.0f, minusBounds[2] + 4.0f, minusBounds[3] + 4.0f);
        shader.setUniformVec4("uTintColor", hoverMinus ? 0.42f : 0.26f, hoverMinus ? 0.45f : 0.29f, 0.36f, 1.0f);
        drawRect2D(minusBounds[0], minusBounds[1], minusBounds[2], minusBounds[3]);
        shader.setUniformVec4("uTintColor", 1.0f, 1.0f, 1.0f, 1.0f);
        drawPixelText("-", minusBounds[0] + 15.0f, minusBounds[1] + 11.0f, 2.8f);

        // Thanh trượt Volume (Slider Track & Fill)
        float[] sliderBounds = getVolumeSliderBounds(windowWidth, windowHeight);
        float volRatio = (audioManager != null) ? audioManager.getVolume() : 0.7f;

        shader.setUniformVec4("uTintColor", 0.08f, 0.09f, 0.12f, 1.0f);
        drawRect2D(sliderBounds[0] - 2.0f, sliderBounds[1] - 2.0f, sliderBounds[2] + 4.0f, sliderBounds[3] + 4.0f);

        shader.setUniformVec4("uTintColor", 0.24f, 0.65f, 0.92f, 1.0f);
        drawRect2D(sliderBounds[0], sliderBounds[1], sliderBounds[2] * volRatio, sliderBounds[3]);

        float knobW = 12.0f;
        float knobX = sliderBounds[0] + sliderBounds[2] * volRatio - knobW * 0.5f;
        knobX = Math.max(sliderBounds[0], Math.min(sliderBounds[0] + sliderBounds[2] - knobW, knobX));
        shader.setUniformVec4("uTintColor", 1.0f, 0.95f, 0.50f, 1.0f);
        drawRect2D(knobX, sliderBounds[1] - 5.0f, knobW, sliderBounds[3] + 10.0f);

        // Nút Tăng âm lượng [+]
        float[] plusBounds = getVolumePlusBounds(windowWidth, windowHeight);
        boolean hoverPlus = isPointInRect(mouseX, mouseY, plusBounds);
        shader.setUniformVec4("uTintColor", 0.75f, 0.78f, 0.82f, 1.0f);
        drawRect2D(plusBounds[0] - 2.0f, plusBounds[1] - 2.0f, plusBounds[2] + 4.0f, plusBounds[3] + 4.0f);
        shader.setUniformVec4("uTintColor", hoverPlus ? 0.42f : 0.26f, hoverPlus ? 0.45f : 0.29f, 0.36f, 1.0f);
        drawRect2D(plusBounds[0], plusBounds[1], plusBounds[2], plusBounds[3]);
        shader.setUniformVec4("uTintColor", 1.0f, 1.0f, 1.0f, 1.0f);
        drawPixelText("+", plusBounds[0] + 15.0f, plusBounds[1] + 11.0f, 2.8f);

        // 6. Dòng hướng dẫn bấm ESC lần 2 để thoát game
        String quitHint = "PRESS ESC AGAIN TO QUIT";
        float hintScale = 1.6f;
        float hintW = quitHint.length() * 6.0f * hintScale;
        shader.setUniformVec4("uTintColor", 0.75f, 0.78f, 0.84f, 0.90f);
        drawPixelText(quitHint, (windowWidth - hintW) * 0.5f, panelY + PANEL_HEIGHT - 28.0f, hintScale);
    }

    // =========================================================================
    // CÁC HÀM KIỂM TRA TỌA ĐỘ CLICK CHUỘT TRONG PAUSE MENU (HIT-TESTING)
    // =========================================================================

    private float[] getResumeButtonBounds(int windowWidth, int windowHeight) {
        float panelY = (windowHeight - PANEL_HEIGHT) * 0.5f;
        float bx = (windowWidth - RESUME_BTN_WIDTH) * 0.5f;
        float by = panelY + 72.0f;
        return new float[]{bx, by, RESUME_BTN_WIDTH, RESUME_BTN_HEIGHT};
    }

    private float[] getVolumeMinusBounds(int windowWidth, int windowHeight) {
        float panelY = (windowHeight - PANEL_HEIGHT) * 0.5f;
        float totalRowW = VOL_BTN_SIZE + 12.0f + SLIDER_WIDTH + 12.0f + VOL_BTN_SIZE;
        float startX = (windowWidth - totalRowW) * 0.5f;
        float by = panelY + 168.0f;
        return new float[]{startX, by, VOL_BTN_SIZE, VOL_BTN_SIZE};
    }

    private float[] getVolumeSliderBounds(int windowWidth, int windowHeight) {
        float[] minus = getVolumeMinusBounds(windowWidth, windowHeight);
        float sx = minus[0] + VOL_BTN_SIZE + 12.0f;
        float sy = minus[1] + (VOL_BTN_SIZE - SLIDER_HEIGHT) * 0.5f;
        return new float[]{sx, sy, SLIDER_WIDTH, SLIDER_HEIGHT};
    }

    private float[] getVolumePlusBounds(int windowWidth, int windowHeight) {
        float[] slider = getVolumeSliderBounds(windowWidth, windowHeight);
        float[] minus = getVolumeMinusBounds(windowWidth, windowHeight);
        float px = slider[0] + SLIDER_WIDTH + 12.0f;
        return new float[]{px, minus[1], VOL_BTN_SIZE, VOL_BTN_SIZE};
    }

    private boolean isPointInRect(double mx, double my, float[] rect) {
        return mx >= rect[0] && mx <= (rect[0] + rect[2]) &&
               my >= rect[1] && my <= (rect[1] + rect[3]);
    }

    public boolean isInsideResumeButton(double mouseX, double mouseY, int windowWidth, int windowHeight) {
        return isPointInRect(mouseX, mouseY, getResumeButtonBounds(windowWidth, windowHeight));
    }

    public boolean isInsideVolumeMinus(double mouseX, double mouseY, int windowWidth, int windowHeight) {
        return isPointInRect(mouseX, mouseY, getVolumeMinusBounds(windowWidth, windowHeight));
    }

    public boolean isInsideVolumePlus(double mouseX, double mouseY, int windowWidth, int windowHeight) {
        return isPointInRect(mouseX, mouseY, getVolumePlusBounds(windowWidth, windowHeight));
    }

    public float getSliderVolumeAtCursor(double mouseX, double mouseY, int windowWidth, int windowHeight) {
        float[] s = getVolumeSliderBounds(windowWidth, windowHeight);
        if (mouseX >= s[0] - 6.0f && mouseX <= s[0] + s[2] + 6.0f &&
            mouseY >= s[1] - 10.0f && mouseY <= s[1] + s[3] + 10.0f) {
            float ratio = (float) ((mouseX - s[0]) / s[2]);
            return Math.max(0.0f, Math.min(1.0f, ratio));
        }
        return -1.0f;
    }

    // =========================================================================
    // CÁC HÀM VẼ HÌNH HỌC CƠ SỞ (2D RECTANGLE, 3D BOX, 5x7 PIXEL FONT)
    // =========================================================================

    private void drawRect2D(float x, float y, float w, float h) {
        drawTexturedRect2D(x, y, w, h, 0.0f, 0.0f, 1.0f, 1.0f);
    }

    private void drawTexturedRect2D(float x, float y, float w, float h, float u0, float v0, float u1, float v1) {
        float[] verts = new float[]{
                x,     y + h, 0.0f, u0, v1, 1.0f,
                x + w, y + h, 0.0f, u1, v1, 1.0f,
                x + w, y,     0.0f, u1, v0, 1.0f,

                x,     y + h, 0.0f, u0, v1, 1.0f,
                x + w, y,     0.0f, u1, v0, 1.0f,
                x,     y,     0.0f, u0, v0, 1.0f
        };
        uploadAndDrawTriangles(verts);
    }

    private void draw3DBoxSolid(float x0, float y0, float z0, float x1, float y1, float z1) {
        float[] verts = new float[]{
                // Top (+Y)
                x0, y1, z1, 0, 0, 1,  x1, y1, z1, 0, 0, 1,  x1, y1, z0, 0, 0, 1,
                x0, y1, z1, 0, 0, 1,  x1, y1, z0, 0, 0, 1,  x0, y1, z0, 0, 0, 1,

    private void carveLiquidLake(

                if (isInsideWorld(x, 1, z)) {
                    float dx = x - centerX;
                    float dz = z - centerZ;
                    if (dx * dx + dz * dz <= radius * radius) {
                        liquidLevel = Math.min(liquidLevel, surfaceHeights[x][z]);
                    }
                }
            }
        }

        liquidLevel = Math.max(4, Math.min(MAX_TERRAIN_Y - 1, liquidLevel));

        for (int x = centerX - rCeil - 1; x <= centerX + rCeil + 1; x++) {
            for (int z = centerZ - rCeil - 1; z <= centerZ + rCeil + 1; z++) {

                    continue;
                }

                float dx = x - centerX;
                float dz = z - centerZ;
                float waveRadius = radius + (float) (Math.sin(x * 0.7) * Math.cos(z * 0.7) * 0.85);
                float dist = (float) Math.sqrt(dx * dx + dz * dz);

                if (dist <= waveRadius) {
                    float normalizedDist = dist / Math.max(1.0f, waveRadius);
                    int localDepth = Math.max(1, Math.round(depth * (1.0f - normalizedDist * 0.65f)));
                    int bottomY = Math.max(2, liquidLevel - localDepth + 1);

                    for (int y = liquidLevel + 1; y <= MAX_TERRAIN_Y + 4; y++) {
                        setBlockInternal(x, y, z, Block.AIR.getId());
                    }

                    for (int y = bottomY; y <= liquidLevel; y++) {
                        setBlockInternal(x, y, z, liquidType.getId());
                    }

                    if (bottomY - 1 > 0) {
                        setBlockInternal(x, bottomY - 1, z, basinBlockType.getId());
                    }
                    surfaceHeights[x][z] = liquidLevel;
                } else if (dist <= waveRadius + 1.3f) {
                    for (int y = 1; y <= liquidLevel; y++) {
                        byte existing = getBlock(x, y, z);
                        if (existing == Block.AIR.getId()) {
                            Block topShore = (liquidType == Block.LAVA)
                                    ? Block.STONE
                                    : (y == liquidLevel ? Block.GRASS : Block.DIRT);
                            setBlockInternal(x, y, z, topShore.getId());
                        }
                    }
                    if (surfaceHeights[x][z] > liquidLevel + 1) {
                        for (int y = liquidLevel + 2; y <= surfaceHeights[x][z]; y++) {
                            setBlockInternal(x, y, z, Block.AIR.getId());
                        }
                        setBlockInternal(x, liquidLevel + 1, z,

                        surfaceHeights[x][z] = liquidLevel + 1;
                    }
                }
            }

    private void generateOakTree(int baseX, int baseY, int baseZ, int trunkHeight, Random rng) {
        if (baseY - 1 > 0) {
            setBlockInternal(baseX, baseY - 1, baseZ, Block.DIRT.getId());
        }

        int topTrunkY = Math.min(WORLD_SIZE_Y - 3, baseY + trunkHeight - 1);

        int leafStart = topTrunkY - 2;
        int leafEnd = topTrunkY + 1;

        for (int ly = leafStart; ly <= leafEnd; ly++) {
            int dy = ly - topTrunkY;
            int radius = (dy <= -1) ? 2 : 1;

            for (int lx = baseX - radius; lx <= baseX + radius; lx++) {
                for (int lz = baseZ - radius; lz <= baseZ + radius; lz++) {
                    int distX = Math.abs(lx - baseX);
                    int distZ = Math.abs(lz - baseZ);

                    if (distX == radius && distZ == radius) {
                        if (dy == 1 || rng.nextBoolean()) {
                            continue;
                        }
                    }

                    if (isInsideWorld(lx, ly, lz) && getBlock(lx, ly, lz) == Block.AIR.getId()) {
                        setBlockInternal(lx, ly, lz, Block.OAK_LEAVES.getId());
                    }
                }
            }
        }

        for (int y = baseY; y <= topTrunkY; y++) {
            if (isInsideWorld(baseX, y, baseZ)) {
                setBlockInternal(baseX, y, baseZ, Block.OAK_LOG.getId());
            }
        }
    }

    private int findHighestBlockY(int wx, int wz) {
        for (int y = WORLD_SIZE_Y - 1; y >= 0; y--) {
            if (getBlock(wx, y, wz) != Block.AIR.getId()) {
                return y;
            }
        }
        return 0;
    }

    private boolean hasLiquidNearby(int wx, int wy, int wz, int radius) {
        for (int x = wx - radius; x <= wx + radius; x++) {
            for (int z = wz - radius; z <= wz + radius; z++) {
                for (int y = wy - 1; y <= wy + 1; y++) {
                    Block t = Block.fromId(getBlock(x, y, z));
                    if (t.isLiquid()) {
                        return true;
                    }
                }
            }
        }
        return false;
    }

        int lx = worldX % Chunk.SIZE;
        int ly = worldY % Chunk.SIZE;
        int lz = worldZ % Chunk.SIZE;

        chunks[cx][cy][cz].setBlockLocal(lx, ly, lz, blockId);
    }

    public boolean isInsideWorld(int worldX, int worldY, int worldZ) {
        return worldX >= 0 && worldX < WORLD_SIZE_X &&
               worldY >= 0 && worldY < WORLD_SIZE_Y &&
               worldZ >= 0 && worldZ < WORLD_SIZE_Z;
    }

    public byte getBlock(int worldX, int worldY, int worldZ) {
        if (!isInsideWorld(worldX, worldY, worldZ)) {
            return 0;
        }
        int cx = worldX / Chunk.SIZE;
        int cy = worldY / Chunk.SIZE;
        int cz = worldZ / Chunk.SIZE;

        int lx = worldX % Chunk.SIZE;
        int ly = worldY % Chunk.SIZE;
        int lz = worldZ % Chunk.SIZE;

        return chunks[cx][cy][cz].getBlockLocal(lx, ly, lz);
    }

    public boolean isSolidBlock(int worldX, int worldY, int worldZ) {
        Block block = Block.fromId(getBlock(worldX, worldY, worldZ));
        return block.isSolid();
    }

    public void setBlock(int worldX, int worldY, int worldZ, byte blockId) {
        if (!isInsideWorld(worldX, worldY, worldZ)) {
            return;
        }

        int cx = worldX / Chunk.SIZE;
        int cy = worldY / Chunk.SIZE;
        int cz = worldZ / Chunk.SIZE;

        int lx = worldX % Chunk.SIZE;
        int ly = worldY % Chunk.SIZE;
        int lz = worldZ % Chunk.SIZE;

        chunks[cx][cy][cz].setBlockLocal(lx, ly, lz, blockId);

        if (lx == 0 && cx > 0) {
            chunks[cx - 1][cy][cz].markDirty();
        } else if (lx == Chunk.SIZE - 1 && cx < WORLD_CHUNKS_X - 1) {
            chunks[cx + 1][cy][cz].markDirty();
        }

        if (ly == 0 && cy > 0) {
            chunks[cx][cy - 1][cz].markDirty();
        } else if (ly == Chunk.SIZE - 1 && cy < WORLD_CHUNKS_Y - 1) {
            chunks[cx][cy + 1][cz].markDirty();
        }

        if (lz == 0 && cz > 0) {
            chunks[cx][cy][cz - 1].markDirty();
        } else if (lz == Chunk.SIZE - 1 && cz < WORLD_CHUNKS_Z - 1) {
            chunks[cx][cy][cz + 1].markDirty();
        }
    }

    /**
     * Đập vỡ hoàn toàn một Block tại `(worldX, worldY, worldZ)` và kích hoạt hành vi đa hình `target.onBreak(...)`.
     *
     * @param worldX tọa độ X thế giới
     * @param worldY tọa độ Y thế giới
     * @param worldZ tọa độ Z thế giới
     */
    public void breakBlockAndSpawnDrop(int worldX, int worldY, int worldZ) {
        Block target = Block.fromId(getBlock(worldX, worldY, worldZ));
        if (!target.isBreakable()) {
            return;
        }

        // 1. Đổi ô tại vị trí đó thành Không khí (Block.AIR)
        setBlock(worldX, worldY, worldZ, Block.AIR.getId());

        // 2. Kích hoạt đa hình hành vi khi vỡ (GrassBlock rơi ra Block.DIRT, các khối khác rơi ra chính nó)
        target.onBreak(this, worldX, worldY, worldZ);
    }

    private void rebuildAllDirtyChunks() {
        for (int cx = 0; cx < WORLD_CHUNKS_X; cx++) {
            for (int cy = 0; cy < WORLD_CHUNKS_Y; cy++) {
                for (int cz = 0; cz < WORLD_CHUNKS_Z; cz++) {
                    Chunk chunk = chunks[cx][cy][cz];
                    if (chunk.isDirty()) {
                        chunk.rebuildMesh(this);
                    }
                }
            }
        }
    }

    /**
     * Vẽ toàn bộ các Chunk địa hình và duyệt danh sách đa hình `entities` để vẽ các thực thể 3D.
     *
     * @param shader chương trình Shader đang kích hoạt
     */
    public void render(ShaderProgram shader) {
        rebuildAllDirtyChunks();

        shader.setUniformMatrix4f("uModel", identityMatrix);
        for (int cx = 0; cx < WORLD_CHUNKS_X; cx++) {
            for (int cy = 0; cy < WORLD_CHUNKS_Y; cy++) {
                for (int cz = 0; cz < WORLD_CHUNKS_Z; cz++) {
                    chunks[cx][cy][cz].render(shader);
                }
            }
        }

        // Duyệt danh sách đa hình List<Entity> và gọi entity.render(shader)
        for (Entity entity : entities) {
            entity.render(shader);
        }

        shader.setUniformMatrix4f("uModel", identityMatrix);
    }

    @Override
    public void dispose() {
        for (int cx = 0; cx < WORLD_CHUNKS_X; cx++) {
            for (int cy = 0; cy < WORLD_CHUNKS_Y; cy++) {
                for (int cz = 0; cz < WORLD_CHUNKS_Z; cz++) {
                    if (chunks[cx][cy][cz] != null) {
                        chunks[cx][cy][cz].dispose();
                    }
                }
            }
        }
        ItemEntity.cleanupSharedMeshes();
    }
}

import com.voxelgame.engine.ShaderProgram;
import com.voxelgame.engine.Texture;
import com.voxelgame.engine.World;

/**
 * Lớp trừu tượng đại diện cho một Khối vuông (Voxel Block), kế thừa từ {@link Entity}
 * và áp dụng mẫu thiết kế Flyweight (Flyweight Design Pattern) để tối ưu hóa bộ nhớ RAM.
 *
 * LƯU Ý KỸ THUẬT QUAN TRỌNG VỀ THỨ TỰ KHỞI TẠO STATIC (STATIC INITIALIZATION ORDER):
 * - Mảng tra cứu `BY_ID` bắt buộc phải được khai báo và cấp phát TRƯỚC các hằng số thực thể
 *   (`AIR`, `DIRT`, `GRASS`...). Khi mỗi hằng số con gọi constructor `super(...)` của `Block`,
 *   nó sẽ tự đăng ký chính mình vào `BY_ID[id & 0xFF] = this;` mà không gặp lỗi NullPointerException.
 */
public abstract class Block extends Entity {

    // 1. BẮT BUỘC KHAI BÁO BẢNG TRA CỨU FLYWEIGHT TRƯỚC TIÊN
    private static final Block[] BY_ID = new Block[16];

    // 2. KHỞI TẠO 9 HẰNG SỐ THỰC THỂ BLOCK DUY NHẤT (FLYWEIGHT SINGLETON INSTANCES)
    public static final Block AIR = new AirBlock();
    public static final Block DIRT = new DirtBlock();
    public static final Block GRASS = new GrassBlock();
    public static final Block STONE = new StoneBlock();
    public static final Block BEDROCK = new BedrockBlock();
    public static final Block OAK_LOG = new OakLogBlock();
    public static final Block OAK_LEAVES = new OakLeavesBlock();
    public static final Block WATER = new WaterBlock();
    public static final Block LAVA = new LavaBlock();

    /**
     * Liệt kê 6 hướng mặt của một khối lập phương 3D.
     */
    public enum Face {
        TOP,    // Mặt trên (+Y)
        BOTTOM, // Mặt dưới (-Y)
        NORTH,  // Mặt Bắc (-Z)
        SOUTH,  // Mặt Nam (+Z)
        EAST,   // Mặt Đông (+X)
        WEST    // Mặt Tây (-X)
    }

    // Các thuộc tính bất biến được đóng gói chặt chẽ (private final)
    private final byte id;
    private final String displayName;
    private final float hardness;      // Thời gian đập vỡ bằng tay không (giây). Âm (-1) là không thể phá hủy
    private final boolean solid;       // Có chặn va chạm vật lý AABB và tia Raycast không?
    private final boolean liquid;      // Có phải khối chất lỏng (Water / Lava) không?
    private final boolean opaque;      // Có che khuất hoàn toàn các mặt khối liền kề không?
    private final float[] iconColor;   // Màu RGB đại diện

    protected Block(
            byte id,
            String displayName,
            float hardness,
            boolean solid,
            boolean liquid,
            boolean opaque,
            float[] iconColor
    ) {
        super(0.5f, 1.0f);
        this.id = id;
        this.displayName = displayName;
        this.hardness = hardness;
        this.solid = solid;
        this.liquid = liquid;
        this.opaque = opaque;
        this.iconColor = (iconColor != null) ? iconColor.clone() : new float[]{0.0f, 0.0f, 0.0f};

        // Tự động đăng ký thực thể Flyweight vào bảng tra cứu O(1)
        BY_ID[id & 0xFF] = this;
    }

    public static Block fromId(byte id) {

        if (index >= 0 && index < BY_ID.length && BY_ID[index] != null) {

        }

    }

     * @param y tọa độ nguyên trục Y của khối bị phá

     * Mặc định các khối rắn có chiều cao `1.0f`. Các khối chất lỏng (`WaterBlock`, `LavaBlock`)
     * ghi đè phương thức này để hạ mặt thoáng xuống `0.88f` khi phía trên là không khí.
     *
     * @param world thế giới voxel
     * @param wx tọa độ X thế giới
     * @param wy tọa độ Y thế giới
     * @param wz tọa độ Z thế giới
     * @return chiều cao thực tế của mặt trên khối (0.0f đến 1.0f)

    public float getTopHeight(World world, int wx, int wy, int wz) {
        return 1.0f;
    }

     * Tính tọa độ chất liệu UV [u0, v0, u1, v1] trên Texture Atlas cho một mặt cụ thể của Block.
     *
     * @param face mặt lập phương
     * @return mảng 4 phần tử float [u0, v0, u1, v1]
     */
    public float[] getFaceUV(Face face) {
        return getTileUV(getTileIndex(face));
    }

    /**
     * Tính trực tiếp tọa độ UV [u0, v0, u1, v1] từ chỉ số ô `tileIndex` trên Texture Atlas 4x4.
     * Áp dụng khoảng lùi eps = 0.0008f ở mép để triệt tiêu hiện tượng lem viền (Texture Bleeding).
     *
     * @param tileIndex chỉ số ô tile (0..15)
     * @return mảng 4 phần tử float [u0, v0, u1, v1]
     */
    public static float[] getTileUV(int tileIndex) {
        int col = tileIndex % Texture.ATLAS_COLS;
        int row = tileIndex / Texture.ATLAS_COLS;

        float tileWidth = 1.0f / Texture.ATLAS_COLS;
        float tileHeight = 1.0f / Texture.ATLAS_ROWS;

        // Khoảng lùi siêu nhỏ ở mép để triệt tiêu hiện tượng lem viền (Texture Bleeding) giữa các ô Atlas
        float eps = 0.0008f;

        float u0 = col * tileWidth + eps;
        float v0 = row * tileHeight + eps;
        float u1 = (col + 1) * tileWidth - eps;
        float v1 = (row + 1) * tileHeight - eps;

        return new float[]{u0, v0, u1, v1};
    }

    public byte getId() {
        return id;
    }

    public String getDisplayName() {
        return displayName;
    }

    public float getHardness() {
        return hardness;
    }

     * Khối rắn có độ cứng > 0 mới có thể đập vỡ.
     *
     * @return true nếu khối có thể bị đập vỡ
     */
    public boolean isBreakable() {
        return solid && hardness > 0.0f;
    }

    public boolean isSolid() {
        return solid;
    }

    public boolean isLiquid() {
        return liquid;
    }

    public boolean isOpaque() {
        return opaque;
    }

    public float[] getIconColor() {
        return iconColor.clone();
    }

package com.voxelgame.entity;

import com.badlogic.gdx.math.Vector3;
import com.voxelgame.engine.ShaderProgram;
import com.voxelgame.engine.World;

/**
 * Lớp cha trừu tượng gốc (Abstract Base Class) đại diện cho mọi thực thể tồn
 * tại trong không gian 3D.
 *
 * THỂ HIỆN 4 TÍNH CHẤT OOP:
 * 1. Tính Trừu tượng (Abstraction):
 * - Khai báo khái niệm tổng quát về một thực thể trong thế giới Voxel với 2
 * hành vi trừu tượng
 * bắt buộc các lớp con định nghĩa: `update(World, float)` và
 * `render(ShaderProgram)`.
 * 2. Tính Kế thừa (Inheritance):
 * - Cung cấp các thuộc tính vật lý cốt lõi (`position`, `velocity`,
 * `halfWidth`, `height`, `onGround`)
 * và thuật toán kiểm tra giao cắt hộp va chạm AABB (`intersectsBlock`) cho mọi
 * lớp con
 * (`Player`, `ItemEntity`, `Block` và 9 lớp Block cụ thể).
 * 3. Tính Đa hình (Polymorphism):
 * - Đa hình lúc biên dịch (Compile-time Polymorphism / Method Overloading): Nạp
 * chồng 2 constructor
 * cho thực thể động có tọa độ khởi tạo và thực thể tĩnh mẫu (Flyweight Block).
 * - Đa hình lúc chạy (Runtime Polymorphism / Method Overriding): Cho phép
 * `World` quản lý chung
 * danh sách `List<Entity>` và gọi `entity.update(...)`, `entity.render(...)`
 * đồng nhất.
 * 4. Tính Đóng gói (Encapsulation):
 * - Bảo vệ trạng thái tọa độ nội bộ bằng kỹ thuật sao chép phòng thủ (Defensive
 * Copying) trong
 * `getPosition()`, `getCenterPosition()` và `getVelocity()`, ngăn bên ngoài sửa
 * trực tiếp vector gốc.
 */
public abstract class Entity {

    // Tọa độ điểm chính giữa dưới đáy (Bottom-Center) của hộp va chạm AABB (sử dụng
    // Vector3 của libGDX)
    protected final Vector3 position;
    // Vector vận tốc chuyển động theo 3 trục (X, Y, Z)
    protected final Vector3 velocity;

    // Một nửa chiều rộng (bán kính ngang) và toàn bộ chiều cao của hộp va chạm AABB
    protected final float halfWidth;
    protected final float height;

    // Cờ trạng thái báo hiệu thực thể đang đứng chạm mặt đất rắn
    protected boolean onGround = false;

    /**
     * Constructor đầy đủ dành cho các thực thể động xuất hiện tại tọa độ cụ thể
     * trong thế giới
     * (ví dụ: Player hoặc ItemEntity văng ra khi đập vỡ Block).
     *
     * @param x         tọa độ khởi tạo trục X
     * @param y         tọa độ khởi tạo trục Y (đáy thực thể)
     * @param z         tọa độ khởi tạo trục Z
     * @param halfWidth một nửa chiều rộng (bán kính theo trục ngang)
     * @param height    tổng chiều cao thực thể theo trục đứng
     */
    protected Entity(float x, float y, float z, float halfWidth, float height) {
        this.position = new Vector3(x, y, z);
        this.velocity = new Vector3(0.0f, 0.0f, 0.0f);
        this.halfWidth = halfWidth;
        this.height = height;
    }

    /**
     * Constructor nạp chồng (Overloaded Constructor) dành cho các thực thể mẫu
     * không gắn cố định
     * với một vị trí duy nhất (ví dụ: các thực thể Flyweight của lớp Block).
     *
     * @param halfWidth một nửa chiều rộng
     * @param height    chiều cao
     */
    protected Entity(float halfWidth, float height) {
        this(0.0f, 0.0f, 0.0f, halfWidth, height);
    }

    /**
     * Kiểm tra xem hộp va chạm AABB của thực thể hiện tại có giao cắt với ô lưới
     * Block
     * tại tọa độ nguyên (blockX, blockY, blockZ) hay không.
     * Dùng để ngăn người chơi đặt Block mới đè vào chính cơ thể mình.
     *
     * @param blockX tọa độ X nguyên của block cần kiểm tra
     * @param blockY tọa độ Y nguyên của block cần kiểm tra
     * @param blockZ tọa độ Z nguyên của block cần kiểm tra
     * @return true nếu hộp va chạm của thực thể giao cắt với khối lập phương của
     *         block
     */

package com.voxelgame.entity;

/**
 * Quản lý hệ thống 9 ô Thanh Công Cụ Nhanh (Hotbar HUD) thuộc sở hữu của {@link Player} (Composition).
 *
 * THỂ HIỆN TÍNH ĐÓNG GÓI (ENCAPSULATION):
 * - Toàn bộ mảng lưu trữ loại khối (`slotTypes`) và số lượng (`slotCounts`) được khai báo `private final`,
 *   ngăn chặn mọi truy cập hoặc chỉnh sửa trực tiếp từ bên ngoài làm sai lệch giới hạn cộng dồn (64 item/ô).
 */
public class Hotbar {

    // Tổng số ô trên thanh Hotbar
    public static final int SLOT_COUNT = 9;
    // Số lượng cộng dồn tối đa trong 1 ô (64 blocks/stack chuẩn Minecraft)
    public static final int MAX_STACK_SIZE = 64;

    private final Block[] slotTypes = new Block[SLOT_COUNT];
    private final int[] slotCounts = new int[SLOT_COUNT];

    // Chỉ số ô đang được người chơi chọn hiện tại (0..8, tương ứng phím 1..9)
    private int selectedSlot = 0;

    public Hotbar() {
        for (int i = 0; i < SLOT_COUNT; i++) {
            slotTypes[i] = Block.AIR;
            slotCounts[i] = 0;
        }

        // Trang bị sẵn 5 loại Block cơ bản ở 5 ô đầu để người chơi thử nghiệm ngay khi vào thế giới
        setSlot(0, Block.DIRT, 16);
        setSlot(1, Block.GRASS, 16);
        setSlot(2, Block.STONE, 16);
        setSlot(3, Block.OAK_LOG, 12);
        setSlot(4, Block.OAK_LEAVES, 16);
    }

    /**
     * Thiết lập trực tiếp loại Block và số lượng cho một ô cụ thể.
     *
     * @param slotIndex chỉ số ô (0..8)
     * @param type loại Block
     * @param count số lượng (tối đa 64)
     */
    public void setSlot(int slotIndex, Block type, int count) {
        if (slotIndex < 0 || slotIndex >= SLOT_COUNT) {
            return;
        }
        if (type == null || type == Block.AIR || count <= 0) {
            slotTypes[slotIndex] = Block.AIR;
            slotCounts[slotIndex] = 0;
        } else {
            slotTypes[slotIndex] = type;
            slotCounts[slotIndex] = Math.min(MAX_STACK_SIZE, count);
        }
    }

    /**
     * Chọn ô trực tiếp theo chỉ số (0 đến 8, tương ứng khi bấm phím 1 đến 9).
     *
     * @param slotIndex chỉ số ô cần chọn (0..8)
     */
    public void selectSlot(int slotIndex) {
        if (slotIndex >= 0 && slotIndex < SLOT_COUNT) {
            this.selectedSlot = slotIndex;
        }
    }

    /**
     * Chuyển đổi ô đang chọn khi người chơi cuộn con lăn chuột (Mouse Scroll Callback).
     *
     * @param scrollYOffset lượng cuộn trục Y (dương: cuộn lên, âm: cuộn xuống)
     */
    public void scroll(double scrollYOffset) {
        if (Math.abs(scrollYOffset) < 0.01) {
            return;
        }
        int step = (scrollYOffset > 0) ? -1 : 1;
        this.selectedSlot = (this.selectedSlot + step + SLOT_COUNT) % SLOT_COUNT;
    }

    /**
     * Thêm một số lượng Block vừa nhặt được (từ ItemEntity rơi ra trên mặt đất) vào thanh Hotbar.
     * Ưu tiên gộp vào các ô cùng loại chưa đầy 64, sau đó mới xếp vào ô trống đầu tiên.
     *
     * @param type loại Block nhặt được
     * @param amount số lượng cần thêm
     * @return true nếu đã thêm được ít nhất 1 item vào Hotbar
     */
    public boolean addItem(Block type, int amount) {
        if (type == null || type == Block.AIR || amount <= 0) {
            return false;
        }

        int remaining = amount;

        // Bước 1: Ưu tiên cộng dồn vào các ô đang chứa sẵn cùng loại Block và chưa đầy 64
        for (int i = 0; i < SLOT_COUNT; i++) {
            if (slotTypes[i] == type && slotCounts[i] < MAX_STACK_SIZE) {
                int spaceLeft = MAX_STACK_SIZE - slotCounts[i];
                int toAdd = Math.min(spaceLeft, remaining);
                slotCounts[i] += toAdd;
                remaining -= toAdd;
                if (remaining == 0) {
                    return true;
                }
            }
        }

        // Bước 2: Nếu vẫn còn dư, đặt vào ô trống đầu tiên từ trái sang phải
        for (int i = 0; i < SLOT_COUNT; i++) {
            if (slotTypes[i] == Block.AIR || slotCounts[i] <= 0) {
                slotTypes[i] = type;
                int toAdd = Math.min(MAX_STACK_SIZE, remaining);
                slotCounts[i] = toAdd;
                remaining -= toAdd;
                if (remaining == 0) {
                    return true;
                }
            }
        }

        return remaining < amount;
    }

    /**
     * Tiêu thụ 1 Block ở ô đang chọn khi người chơi click chuột phải để đặt Block.
     *
     * @return loại Block vừa tiêu thụ, hoặc Block.AIR nếu ô trống
     */
    public Block consumeSelectedItem() {
        Block currentType = slotTypes[selectedSlot];
        int currentCount = slotCounts[selectedSlot];

        if (currentType == Block.AIR || currentCount <= 0) {
            return Block.AIR;
        }

        slotCounts[selectedSlot] = currentCount - 1;
        if (slotCounts[selectedSlot] <= 0) {
            slotCounts[selectedSlot] = 0;
            slotTypes[selectedSlot] = Block.AIR;
        }

        return currentType;
    }

    public int getSelectedSlot() {
        return selectedSlot;
    }

    public Block getSelectedBlock() {
        return slotTypes[selectedSlot];
    }

    public Block getSelectedBlockType() {
        return slotTypes[selectedSlot];
    }

    public int getSelectedCount() {
        return slotCounts[selectedSlot];
    }

    public Block getSlotBlock(int slotIndex) {
        if (slotIndex < 0 || slotIndex >= SLOT_COUNT) {
            return Block.AIR;
        }
        return slotTypes[slotIndex];
    }

    public Block getSlotType(int slotIndex) {
        return getSlotBlock(slotIndex);
    }

    public int getSlotCount(int slotIndex) {
        if (slotIndex < 0 || slotIndex >= SLOT_COUNT) {
            return 0;
        }
        return slotCounts[slotIndex];
    }
}

package com.voxelgame.entity;

import com.badlogic.gdx.graphics.GL20;
import com.badlogic.gdx.graphics.Mesh;
import com.badlogic.gdx.graphics.VertexAttribute;
import com.badlogic.gdx.graphics.VertexAttributes;
import com.badlogic.gdx.graphics.VertexAttributes.Usage;

import com.badlogic.gdx.math.Vector3;
import com.voxelgame.engine.ShaderProgram;
import com.voxelgame.engine.World;

import java.util.Map;

/**
 * Thực thể Vật phẩm 3D nhỏ rơi ra ngoài thế giới sau khi người chơi đập vỡ một Block, kế thừa từ {@link Entity}.
 *
 * THỂ HIỆN CÁC TÍNH CHẤT OOP:
 * 1. Kế thừa (Inheritance): Kế thừa `position`, `velocity`, `halfWidth`, `height`, `onGround` từ `Entity`.
 * 2. Đa hình lúc biên dịch (Method Overloading): Cung cấp 2 constructor nạp chồng — một tự sinh lực văng
 *    ngẫu nhiên và một nhận vector vận tốc khởi tạo tùy chỉnh.
 * 3. Đa hình lúc chạy (Method Overriding): Ghi đè `update(World, float)` và `render(ShaderProgram)` của `Entity`.
 * 4. Đóng gói & Quản lý bộ nhớ GPU: Sử dụng `Map<Block, Mesh> SHARED_MESHES = new HashMap<>()` để dùng chung
 *    lưới đa giác Mesh cho từng loại `Block` thay vì cấp phát trùng lặp trên VRAM.
 */
public class ItemEntity extends Entity {

    // Kích thước cạnh của khối Item thu nhỏ (0.28 block ~ bằng 1/3 kích thước khối thật)
    private static final float ITEM_SCALE = 0.28f;
    private static final float HALF_SIZE = ITEM_SCALE * 0.5f;

    // Các hằng số vật lý của Item rơi
    private static final float ITEM_GRAVITY = -18.0f;
    private static final float PICKUP_DELAY = 0.25f;
    private static final float MAGNET_DISTANCE = 1.85f;
    private static final float PICKUP_DISTANCE = 0.85f;
    private static final float MAX_LIFETIME = 300.0f;

    private final Block block;
    private float age = 0.0f;
    private boolean collected = false;

    private final Matrix4 modelMatrix = new Matrix4();

    // Bộ nhớ đệm lưu trữ lưới Mesh dùng chung cho từng loại Block
    private static final Map<Block, Mesh> SHARED_MESHES = new HashMap<>();

    // Thuộc tính đỉnh của Mesh ItemEntity: aPos (3 floats), aTexCoord (2 floats), aShade (1 float)
    private static final VertexAttributes VERTEX_ATTRIBUTES = new VertexAttributes(
            new VertexAttribute(Usage.Position, 3, "aPos"),
            new VertexAttribute(Usage.TextureCoordinates, 2, "aTexCoord"),
            new VertexAttribute(Usage.Generic, 1, "aShade")
    );

    /**
     * Constructor 1: Tự động tạo vận tốc nảy nhẹ ngẫu nhiên khi khối vừa vỡ ra.
     *
     * @param startX tọa độ spawn trục X
     * @param startY tọa độ spawn trục Y
     * @param startZ tọa độ spawn trục Z
     * @param block loại khối mà vật phẩm đại diện
     */
    public ItemEntity(float startX, float startY, float startZ, Block block) {
        this(
                startX,
                startY,
                startZ,
                new Vector3(
                        (float) ((Math.random() - 0.5) * 1.4),
                        3.2f,
                        (float) ((Math.random() - 0.5) * 1.4)
                ),
                block
        );
    }

    /**
     * Constructor 2 (Overloaded Constructor): Cho phép chỉ định vector vận tốc văng cụ thể.
     *
     * @param startX tọa độ spawn trục X
     * @param startY tọa độ spawn trục Y
     * @param startZ tọa độ spawn trục Z
     * @param initialVelocity vector vận tốc văng ban đầu
     * @param block loại khối mà vật phẩm đại diện
     */
    public ItemEntity(float startX, float startY, float startZ, Vector3 initialVelocity, Block block) {
        super(startX, startY, startZ, HALF_SIZE, ITEM_SCALE);
        if (initialVelocity != null) {
            this.velocity.set(initialVelocity);
        }
        this.block = (block != null) ? block : Block.AIR;
    }

    /**
     * Ghi đè phương thức trừu tượng `update` của `Entity`:
     * Cập nhật vật lý rơi, hiệu ứng lực hút nam châm về phía `Player` và tự động cộng vào `Hotbar`.
     */
    @Override
    public void update(World world, float deltaTime) {
        if (world == null) {
            return;
        }

        Player player = world.getPlayer();
        Hotbar hotbar = (player != null) ? player.getHotbar() : null;
        update(world, player, hotbar, deltaTime);
    }

    /**
     * Phương thức nạp chồng hỗ trợ cập nhật trực tiếp với đối tượng `Player` và `Hotbar`.
     *
     * @param world thế giới voxel
     * @param player người chơi
     * @param hotbar thanh công cụ của người chơi
     * @param deltaTime thời gian giữa hai khung hình
     */
    public void update(World world, Player player, Hotbar hotbar, float deltaTime) {
        if (collected) {
            return;
        }

        age += deltaTime;
        if (age >= MAX_LIFETIME) {
            collected = true;
            return;
        }

        // 1. Kiểm tra khoảng cách tới tâm cơ thể người chơi để kích hoạt lực hút nam châm
        if (player != null && hotbar != null) {
            Vector3 playerCenter = player.getCenterPosition();
            float distSq = position.dst2(playerCenter);

            if (age >= PICKUP_DELAY && distSq <= MAGNET_DISTANCE * MAGNET_DISTANCE) {
                if (distSq <= PICKUP_DISTANCE * PICKUP_DISTANCE) {
                    if (hotbar.addItem(block, 1)) {
                        collected = true;
                        return;
                    }
                }

                Vector3 pullDir = new Vector3(playerCenter).sub(position);
                if (pullDir.len2() > 0.0001f) {
                    pullDir.nor().scl(9.5f);
                    position.add(pullDir.x * deltaTime, pullDir.y * deltaTime, pullDir.z * deltaTime);
                    return;
                }
            }
        }

        // 2. Áp dụng trọng lực kéo xuống và ma sát ngang
        velocity.y += ITEM_GRAVITY * deltaTime;
        if (velocity.y < -25.0f) {
            velocity.y = -25.0f;
        }

        float damping = onGround ? 0.82f : 0.96f;
        velocity.x *= damping;
        velocity.z *= damping;

        // 3. Cập nhật tọa độ và kiểm tra va chạm với các Block rắn trong World
        position.x += velocity.x * deltaTime;
        if (world.isSolidBlock((int) Math.floor(position.x), (int) Math.floor(position.y), (int) Math.floor(position.z))) {
            position.x -= velocity.x * deltaTime;
            velocity.x = 0.0f;
        }

        position.z += velocity.z * deltaTime;
        if (world.isSolidBlock((int) Math.floor(position.x), (int) Math.floor(position.y), (int) Math.floor(position.z))) {
            position.z -= velocity.z * deltaTime;
            velocity.z = 0.0f;
        }

        onGround = false;
        position.y += velocity.y * deltaTime;

        int blockX = (int) Math.floor(position.x);
        int blockYBelow = (int) Math.floor(position.y - HALF_SIZE);
        int blockZ = (int) Math.floor(position.z);

        if (velocity.y <= 0.0f && world.isSolidBlock(blockX, blockYBelow, blockZ)) {
            position.y = blockYBelow + 1.0f + HALF_SIZE;
            velocity.y = 0.0f;
            onGround = true;
        }
    }

    /**
     * Ghi đè phương thức trừu tượng `render` của `Entity`:
     * Vẽ khối 3D thu nhỏ của Item tại vị trí hiện tại kèm hiệu ứng tự xoay tròn và nhấp nhô.
     */
    @Override
    public void render(ShaderProgram shader) {
        if (collected || block == null || block == Block.AIR) {
            return;
        }

        Mesh mesh = getOrCreateMesh(block);

        float bobOffset = (float) Math.sin(age * 3.8f) * 0.07f + 0.08f;
        float rotationDegrees = age * 110.0f;

        modelMatrix.idt()
                .translate(position.x, position.y + bobOffset, position.z)
                .rotate(Vector3.Y, rotationDegrees)
                .scale(ITEM_SCALE, ITEM_SCALE, ITEM_SCALE);

        shader.setUniformMatrix4f("uModel", modelMatrix);
        mesh.render(shader.getGdxShader(), GL20.GL_TRIANGLES);
    }

    public boolean isCollected() {
        return collected;
    }

    public Block getBlock() {
        return block;
    }

    public Block getBlockType() {
        return block;
    }

    private static Mesh getOrCreateMesh(Block type) {
        return SHARED_MESHES.computeIfAbsent(type, t -> {
            float[] topUV = t.getFaceUV(Block.Face.TOP);
            float[] botUV = t.getFaceUV(Block.Face.BOTTOM);
            float[] sideUV = t.getFaceUV(Block.Face.NORTH);

            float[] vertices = new float[216];
            int idx = 0;

            // 1. Mặt Trên (+Y)
            idx = putFace(vertices, idx,
                    -0.5f,  0.5f,  0.5f, topUV[0], topUV[3],
                     0.5f,  0.5f,  0.5f, topUV[2], topUV[3],
                     0.5f,  0.5f, -0.5f, topUV[2], topUV[1],
                    -0.5f,  0.5f, -0.5f, topUV[0], topUV[1], 1.0f);

            // 2. Mặt Dưới (-Y)
            idx = putFace(vertices, idx,
                    -0.5f, -0.5f, -0.5f, botUV[0], botUV[3],
                     0.5f, -0.5f, -0.5f, botUV[2], botUV[3],
                     0.5f, -0.5f,  0.5f, botUV[2], botUV[1],
                    -0.5f, -0.5f,  0.5f, botUV[0], botUV[1], 0.55f);

            // 3. Mặt Trước (+Z)
            idx = putFace(vertices, idx,
                    -0.5f, -0.5f,  0.5f, sideUV[0], sideUV[3],
                     0.5f, -0.5f,  0.5f, sideUV[2], sideUV[3],
                     0.5f,  0.5f,  0.5f, sideUV[2], sideUV[1],
                    -0.5f,  0.5f,  0.5f, sideUV[0], sideUV[1], 0.85f);

            // 4. Mặt Sau (-Z)
            idx = putFace(vertices, idx,
                     0.5f, -0.5f, -0.5f, sideUV[0], sideUV[3],
                    -0.5f, -0.5f, -0.5f, sideUV[2], sideUV[3],
                    -0.5f,  0.5f, -0.5f, sideUV[2], sideUV[1],
                     0.5f,  0.5f, -0.5f, sideUV[0], sideUV[1], 0.85f);

            // 5. Mặt Phải (+X)
            idx = putFace(vertices, idx,
                     0.5f, -0.5f,  0.5f, sideUV[0], sideUV[3],

            float[] arr, int offset,
            float x1, float y1, float z1, float u1, float v1,
            float x2, float y2, float z2, float u2, float v2,

        offset = putVert(arr, offset, x1, y1, z1, u1, v1, shade);
        offset = putVert(arr, offset, x3, y3, z3, u3, v3, shade);
        offset = putVert(arr, offset, x4, y4, z4, u4, v4, shade);
        return offset;
    }

    private static int putVert(float[] arr, int offset, float x, float y, float z, float u, float v, float shade) {
        arr[offset++] = x;
        arr[offset++] = y;
        arr[offset++] = z;
        arr[offset++] = u;
        arr[offset++] = v;
        arr[offset++] = shade;
        return offset;

 *    và quản lý toàn bộ trạng thái chuyển động nội bộ (Bay Double-tap Space, Chạy nhanh Left Ctrl,

    private static final float PLAYER_HEIGHT = 1.8f;

    private static final float SNEAKING_EYE_HEIGHT = 1.45f;

    // Các hằng số tốc độ di chuyển và vật lý
    private static final float MOVE_SPEED = 5.5f;
    private static final float SPRINT_MULTIPLIER = 1.5f;
    private static final float SNEAK_MULTIPLIER = 0.5f;
    private static final float FLY_SPEED_MULTIPLIER = 1.8f;
    private static final float FLY_ASCEND_SPEED = 7.0f;

    private static final float GRAVITY = -26.0f;
    private static final float JUMP_VELOCITY = 8.6f;
    private static final float TERMINAL_VELOCITY = -45.0f;

    // Khoảng thời gian tối đa giữa 2 lần bấm Space để kích hoạt chế độ Bay (0.30 giây)
    private static final float DOUBLE_TAP_THRESHOLD = 0.30f;

    // Quan hệ Hợp thành (Composition - Player "has-a" Hotbar)
    private final Hotbar hotbar = new Hotbar();

    // Tham chiếu góc nhìn Camera
    private Camera camera;

    // Các cờ trạng thái chuyển động của nhân vật
    private boolean isFlying = false;
    private boolean isSprinting = false;
    private boolean isSneaking = false;

    // Biến phục vụ nhận diện thao tác bấm đúp phím Space (Double-tap Space)
    private boolean wasSpacePressed = false;
    private float lastSpaceTapTime = -1.0f;
    private float playerElapsedTime = 0.0f;

    public Player(Vector3 startPosition) {

    public void update(World world, Camera camera, float deltaTime) {

        update(world, deltaTime);

    /**
     * Ghi đè phương thức trừu tượng `update` của `Entity`:
     * Thực thi toàn bộ điều khiển bàn phím, trạng thái Bay/Chạy/Rón rén, vật lý va chạm AABB
     * và đồng bộ vị trí mắt Camera.
     */
    @Override
    public void update(World world, float deltaTime) {
        if (camera == null || world == null) {
            return;
        }

        playerElapsedTime += deltaTime;

        handleInput(world, camera);
        applyPhysicsAndCollision(world, deltaTime);

        // Nếu rơi xuống dưới đáy bản đồ, đưa người chơi trở lại vị trí an toàn trên mặt đất
        if (position.y < -25.0f) {
            position.set(World.WORLD_SIZE_X / 2.0f, 22.0f, World.WORLD_SIZE_Z / 2.0f);

        currentEyeHeight += (targetEye - currentEyeHeight) * eyeLerp;

        // Đồng bộ tọa độ mắt Camera và hiệu ứng góc ống kính (Dynamic FOV)
        camera.setPosition(position.x, position.y + currentEyeHeight, position.z);
        camera.updateDynamicFov(isSprinting, isFlying, deltaTime);
    }

     * Vì trò chơi sử dụng góc nhìn thứ nhất (First-Person Perspective), cơ thể người chơi không cần vẽ.

    private void handleInput(World world, Camera camera) {
        boolean spaceDown = Gdx.input.isKeyPressed(Input.Keys.SPACE);
        boolean shiftDown = Gdx.input.isKeyPressed(Input.Keys.SHIFT_LEFT);
        boolean ctrlDown = Gdx.input.isKeyPressed(Input.Keys.CONTROL_LEFT);

        // 1. Xử lý Double-Tap Space để Bật/Tắt chế độ Bay (Fly Mode)
        if (spaceDown && !wasSpacePressed) {
            float now = playerElapsedTime;
            if (now - lastSpaceTapTime <= DOUBLE_TAP_THRESHOLD) {
                isFlying = !isFlying;
                velocity.y = 0.0f;
                if (isFlying) {
                    onGround = false;
                    isSneaking = false;
                }
                lastSpaceTapTime = -1.0f;
            } else {
                lastSpaceTapTime = now;
            }
        }
        wasSpacePressed = spaceDown;

        // 2. Xử lý phím Left Shift (Thoát chế độ Bay hoặc Đi rón rén Sneak)
        if (isFlying) {
            if (shiftDown) {
                isFlying = false;
                isSneaking = false;
            }
        } else {
            isSneaking = shiftDown;
        }

        // 3. Tính hướng di chuyển trên mặt phẳng ngang XZ
        double yawRad = Math.toRadians(camera.getYaw());
        Vector3 forwardXZ = new Vector3((float) Math.cos(yawRad), 0.0f, (float) Math.sin(yawRad)).nor();
        Vector3 rightXZ = new Vector3(-forwardXZ.z, 0.0f, forwardXZ.x).nor();

        Vector3 wishDir = new Vector3(0.0f, 0.0f, 0.0f);
        if (Gdx.input.isKeyPressed(Input.Keys.W)) {
            wishDir.add(forwardXZ);
        }
        if (Gdx.input.isKeyPressed(Input.Keys.S)) {
            wishDir.sub(forwardXZ);
        }
        if (Gdx.input.isKeyPressed(Input.Keys.D)) {
            wishDir.add(rightXZ);
        }
        if (Gdx.input.isKeyPressed(Input.Keys.A)) {
            wishDir.sub(rightXZ);
        }

        boolean isMovingHorizontally = wishDir.len2() > 0.0001f;

        // 4. Tính tốc độ di chuyển (Sprint 1.5x / Sneak 0.5x / Fly)
        isSprinting = ctrlDown && isMovingHorizontally && !isSneaking;

        float currentSpeed = MOVE_SPEED;
        if (isFlying) {
            currentSpeed *= FLY_SPEED_MULTIPLIER;
            if (isSprinting) {
                currentSpeed *= 1.35f;
            }
        } else if (isSneaking) {
            currentSpeed *= SNEAK_MULTIPLIER;
        } else if (isSprinting) {
            currentSpeed *= SPRINT_MULTIPLIER;
        }

        boolean inLiquid = isInLiquid(world);
        if (inLiquid && !isFlying) {
            currentSpeed *= 0.65f;
        }

        if (isMovingHorizontally) {
            wishDir.nor().scl(currentSpeed);
        }

        velocity.x = wishDir.x;
        velocity.z = wishDir.z;

        // 5. Xử lý chuyển động dọc (Trục Y: Bay lên, Bơi lên hoặc Nhảy)
        if (isFlying) {
            if (spaceDown) {
                velocity.y = FLY_ASCEND_SPEED;
            } else {
                velocity.y = 0.0f;
            }
        } else if (inLiquid) {
            if (spaceDown) {
                velocity.y = 4.2f;
            }
        } else {
            if (spaceDown && onGround) {
                velocity.y = JUMP_VELOCITY;
                onGround = false;
            }
        }
    }

    private void applyPhysicsAndCollision(World world, float deltaTime) {
        if (!isFlying) {
            if (isInLiquid(world)) {
                velocity.y += (GRAVITY * 0.25f) * deltaTime;
                if (velocity.y < -3.5f) {
                    velocity.y = -3.5f;
                }
            } else {
                velocity.y += GRAVITY * deltaTime;
                if (velocity.y < TERMINAL_VELOCITY) {
                    velocity.y = TERMINAL_VELOCITY;
                }
            }
        }

        float dx = velocity.x * deltaTime;
        float dz = velocity.z * deltaTime;

        // Nếu đang giữ Left Shift (Sneak) trên mặt đất, kích hoạt bảo vệ chống rơi khỏi mép block
        if (isSneaking && onGround && !isFlying) {
            dx = preventFallingOffEdgeX(world, dx);
            dz = preventFallingOffEdgeZ(world, dz);
        }

        // Di chuyển và giải quyết va chạm tách trục (X -> Z -> Y)
        position.x += dx;
        resolveCollisionX(world);

        position.z += dz;
        resolveCollisionZ(world);

        onGround = false;
        position.y += velocity.y * deltaTime;
        resolveCollisionY(world);

        if (onGround && isFlying) {
            isFlying = false;
        }
    }

    private float preventFallingOffEdgeX(World world, float dx) {
        float step = 0.02f;
        while (dx != 0.0f && !hasGroundBeneath(world, position.x + dx, position.y, position.z)) {
            if (Math.abs(dx) <= step) {
                dx = 0.0f;
            } else {
                dx += (dx > 0) ? -step : step;
            }
        }
        if (dx == 0.0f) {
            velocity.x = 0.0f;
        }
        return dx;
    }

    private float preventFallingOffEdgeZ(World world, float dz) {
        float step = 0.02f;
        while (dz != 0.0f && !hasGroundBeneath(world, position.x, position.y, position.z + dz)) {
            if (Math.abs(dz) <= step) {
                dz = 0.0f;
            } else {
                dz += (dz > 0) ? -step : step;
            }
        }
        if (dz == 0.0f) {
            velocity.z = 0.0f;
        }
        return dz;
    }

    private boolean hasGroundBeneath(World world, float testX, float testY, float testZ) {
        float checkHalfWidth = halfWidth - 0.05f;
        int minX = (int) Math.floor(testX - checkHalfWidth);
        int maxX = (int) Math.floor(testX + checkHalfWidth);
        int checkY = (int) Math.floor(testY - 0.05f);
        int minZ = (int) Math.floor(testZ - checkHalfWidth);
        int maxZ = (int) Math.floor(testZ + checkHalfWidth);

        for (int x = minX; x <= maxX; x++) {
            for (int z = minZ; z <= maxZ; z++) {
                if (world.isSolidBlock(x, checkY, z)) {
                    return true;
                }
            }
        }
        return false;
    }

    public boolean isInLiquid(World world) {
        int bx = (int) Math.floor(position.x);
        int byFeet = (int) Math.floor(position.y + 0.1f);
        int byWaist = (int) Math.floor(position.y + 0.9f);
        int bz = (int) Math.floor(position.z);

        Block feetBlock = Block.fromId(world.getBlock(bx, byFeet, bz));
        Block waistBlock = Block.fromId(world.getBlock(bx, byWaist, bz));
        return feetBlock.isLiquid() || waistBlock.isLiquid();
    }

    private void resolveCollisionX(World world) {
        int minX = (int) Math.floor(position.x - halfWidth);
        int maxX = (int) Math.floor(position.x + halfWidth);
        int minY = (int) Math.floor(position.y);
        int maxY = (int) Math.floor(position.y + height - 0.0001f);
        int minZ = (int) Math.floor(position.z - halfWidth);
        int maxZ = (int) Math.floor(position.z + halfWidth);

        for (int x = minX; x <= maxX; x++) {
            for (int y = minY; y <= maxY; y++) {
                for (int z = minZ; z <= maxZ; z++) {
                    if (world.isSolidBlock(x, y, z)) {
                        if (velocity.x > 0) {
                            position.x = x - halfWidth - 0.0001f;
                            velocity.x = 0;
                        } else if (velocity.x < 0) {
                            position.x = x + 1.0f + halfWidth + 0.0001f;
                            velocity.x = 0;
                        }
                        return;
                    }
                }
            }
        }
    }

    private void resolveCollisionZ(World world) {
        int minX = (int) Math.floor(position.x - halfWidth);
        int maxX = (int) Math.floor(position.x + halfWidth);
        int minY = (int) Math.floor(position.y);
        int maxY = (int) Math.floor(position.y + height - 0.0001f);
        int minZ = (int) Math.floor(position.z - halfWidth);
        int maxZ = (int) Math.floor(position.z + halfWidth);

        for (int x = minX; x <= maxX; x++) {
            for (int y = minY; y <= maxY; y++) {
                for (int z = minZ; z <= maxZ; z++) {
                    if (world.isSolidBlock(x, y, z)) {
                        if (velocity.z > 0) {
                            position.z = z - halfWidth - 0.0001f;
                            velocity.z = 0;
                        } else if (velocity.z < 0) {
                            position.z = z + 1.0f + halfWidth + 0.0001f;
                            velocity.z = 0;
                        }
                        return;
                    }
                }
            }
        }
    }

    private void resolveCollisionY(World world) {
        int minX = (int) Math.floor(position.x - halfWidth);
        int maxX = (int) Math.floor(position.x + halfWidth);
        int minY = (int) Math.floor(position.y);
        int maxY = (int) Math.floor(position.y + height - 0.0001f);
        int minZ = (int) Math.floor(position.z - halfWidth);
        int maxZ = (int) Math.floor(position.z + halfWidth);

        for (int x = minX; x <= maxX; x++) {
            for (int y = minY; y <= maxY; y++) {
                for (int z = minZ; z <= maxZ; z++) {
                    if (world.isSolidBlock(x, y, z)) {
                        if (velocity.y < 0) {
                            position.y = y + 1.0f;
                            velocity.y = 0;
                            onGround = true;
                        } else if (velocity.y > 0) {
                            position.y = y - height - 0.0001f;
                            velocity.y = 0;
                        }
                        return;
                    }
                }
            }
        }
    }

     *   1. Cập nhật logic game (vật lý, va chạm, đào/đặt block, nhập liệu chuột)

    @Override

        // =================================================================
        float deltaTime = Gdx.graphics.getDeltaTime();
        if (deltaTime > 0.05f) {
            deltaTime = 0.05f;
        }

        // =================================================================
        // BƯỚC 2: Xử lý xoay Camera bằng di chuyển chuột (khi không Pause)
        // =================================================================
        if (!isPaused) {
            processMouseLook();
        }

        // =================================================================
        // BƯỚC 3: Cập nhật logic game (khi không Pause)
        // =================================================================
        Raycast.HitResult currentHit = null;
        float miningProgressRatio = 0.0f;

        if (!isPaused) {
            // Cập nhật đa hình toàn bộ Entity (Player + mọi ItemEntity)
            world.updateEntities(deltaTime);

            // Bắn tia từ mắt Camera theo hướng nhìn để tìm block đang ngắm
            currentHit = Raycast.cast(world, camera.getPosition(), camera.getFront(), RAYCAST_REACH);

            // Cập nhật tiến trình đào block nếu đang giữ chuột trái
            miningProgressRatio = updateMiningProgress(currentHit, deltaTime);
        }

        // =================================================================
        // BƯỚC 4: Xóa bộ đệm màu và bộ đệm chiều sâu
        // =================================================================
        Gdx.gl.glClear(GL20.GL_COLOR_BUFFER_BIT | GL20.GL_DEPTH_BUFFER_BIT);

        // =================================================================
        // BƯỚC 5: Kết xuất Thế giới 3D (Các Chunk địa hình + Các Entity đa hình)
        // =================================================================
        shaderProgram.bind();
        shaderProgram.enable3DWorldMode();

        // Lấy kích thước cửa sổ hiện tại (hỗ trợ resize)
        int screenWidth = Gdx.graphics.getWidth();
        int screenHeight = Gdx.graphics.getHeight();

        // Thiết lập ma trận chiếu phối cảnh (Perspective) và ma trận nhìn (View)
        shaderProgram.setUniformMatrix4f("uProjection", camera.getProjectionMatrix(screenWidth, screenHeight));
        shaderProgram.setUniformMatrix4f("uView", camera.getViewMatrix());

        // Truyền thông số sương mù khoảng cách tuyến tính (Distance Fog)
        shaderProgram.setUniformVec3("uFogColor", SKY_COLOR);
        shaderProgram.setUniform1f("uFogStart", 28.0f);
        shaderProgram.setUniform1f("uFogEnd", 58.0f);

        // Kết xuất các Chunk và toàn bộ Entity trong thế giới
        textureAtlas.bind();
        world.render(shaderProgram);
        textureAtlas.unbind();

        // =================================================================
        // BƯỚC 6: Vẽ hiệu ứng Overlay 3D quanh block đang ngắm/đào
        // =================================================================
        if (!isPaused && currentHit != null && currentHit.isHit()) {
            uiRenderer.renderTargetBlockOverlay3D(shaderProgram, currentHit, miningProgressRatio);
        }

        // =================================================================
        // BƯỚC 7: Kết xuất Giao diện 2D (Crosshair, Hotbar, Thanh đào, Pause Menu)
        // =================================================================
        uiRenderer.render2DHUD(
                shaderProgram,
                textureAtlas,
                camera,
                player.getHotbar(),
                audioManager,
                screenWidth,
                screenHeight,
                miningProgressRatio,
                isPaused,
                currentMouseX,
                currentMouseY
        );

        shaderProgram.unbind();
    }

    /**
     * Xử lý xoay Camera bằng delta chuột mỗi khung hình.
     * Sử dụng {@code Gdx.input.getDeltaX()} và {@code getDeltaY()} để lấy khoảng cách
     * di chuyển chuột tích lũy từ khung hình trước.
     *
     * Kỹ thuật chống giật góc nhìn (Anti-Camera Jumping):
     * Khi vừa bắt chuột lại (sau Pause hoặc lần đầu khởi chạy), cờ {@code skipNextMouseDelta}
     * được bật để bỏ qua delta đột biến do hệ điều hành đưa con trỏ về tâm màn hình.
     */
    private void processMouseLook() {
        if (skipNextMouseDelta) {
            // Bỏ qua delta chuột ở khung hình đầu tiên sau khi bắt lại chuột
            // để tránh giật góc nhìn Camera đột ngột
            skipNextMouseDelta = false;
            return;
        }

        // Lấy khoảng cách di chuyển chuột tích lũy từ khung hình trước
        float deltaX = Gdx.input.getDeltaX();
        float deltaY = Gdx.input.getDeltaY();

        // Áp dụng độ nhạy chuột và xoay Camera
        // deltaX > 0 nghĩa là chuột di chuyển sang phải -> tăng yaw (quay phải)
        // deltaY > 0 trong libGDX nghĩa là chuột di chuyển xuống dưới -> giảm pitch (nhìn xuống)
        // Nên truyền -deltaY để khớp với quy ước "kéo chuột lên = nhìn lên"
        if (camera != null) {
            camera.rotate(deltaX * MOUSE_SENSITIVITY, -deltaY * MOUSE_SENSITIVITY);
        }
    }

    // =====================================================================
    // XỬ LÝ TRẠNG THÁI PAUSE MENU
    // =====================================================================

    /**
     * Chuyển game sang trạng thái Tạm dừng (Pause).
     * - Dừng cập nhật Entity và Raycast
     * - Mở khóa con trỏ chuột để tương tác với Pause Menu
     * - Reset tiến trình đào block đang dở
     */
    private void pauseGame() {
        isPaused = true;
        isLeftMouseDown = false;
        isDraggingVolumeSlider = false;
        resetMiningProgress();

        // Mở khóa con trỏ chuột để người dùng click vào nút Resume, chỉnh Volume
        Gdx.input.setCursorCatched(false);
    }

    /**
     * Tiếp tục game từ trạng thái Tạm dừng.
     * - Bật lại cập nhật Entity và Raycast
     * - Khóa lại con trỏ chuột ở tâm màn hình để xoay Camera
     * - Bật cờ {@code skipNextMouseDelta} để tránh giật góc nhìn Camera
     */
    private void resumeGame() {
        isPaused = false;
        isDraggingVolumeSlider = false;

        // Khóa lại con trỏ chuột để tiếp tục xoay Camera
        Gdx.input.setCursorCatched(true);

        // Bật cờ bỏ qua delta chuột ở khung hình đầu tiên sau khi khóa lại chuột
        // để con trỏ bị kéo về tâm không tạo giật góc nhìn đột ngột
        skipNextMouseDelta = true;
    }

    // =====================================================================

    /**
     * Xử lý sự kiện click chuột trên Pause Menu.
     * Kiểm tra Hit-Testing cho nút Resume, nút giảm/tăng Volume, và thanh trượt Volume.
     *
     * @param screenX tọa độ X chuột trên màn hình (tính từ trái)
     * @param screenY tọa độ Y chuột trên màn hình (tính từ đỉnh, hệ Top-Left của libGDX)
     * @param pressed true nếu vừa nhấn chuột, false nếu vừa nhả chuột
     */
    private void handlePauseMenuMouseClick(float screenX, float screenY, boolean pressed) {
        if (uiRenderer == null) {
            return;
        }

        int screenWidth = Gdx.graphics.getWidth();
        int screenHeight = Gdx.graphics.getHeight();

        // Nhả chuột: dừng kéo thanh trượt Volume
        if (!pressed) {
            isDraggingVolumeSlider = false;
            return;
        }

        // Nhấn chuột: kiểm tra vị trí click

        // Kiểm tra nút "RESUME GAME"
        if (uiRenderer.isInsideResumeButton(screenX, screenY, screenWidth, screenHeight)) {
            resumeGame();
            return;
        }

        // Kiểm tra nút "[-]" giảm âm lượng 10%
        if (uiRenderer.isInsideVolumeMinus(screenX, screenY, screenWidth, screenHeight)) {
            if (audioManager != null) {
                audioManager.changeVolume(-0.10f);
            }
            return;
        }

        // Kiểm tra nút "[+]" tăng âm lượng 10%
        if (uiRenderer.isInsideVolumePlus(screenX, screenY, screenWidth, screenHeight)) {
            if (audioManager != null) {
                audioManager.changeVolume(0.10f);
            }
            return;
        }

        // Kiểm tra click vào thanh trượt Volume -> bắt đầu kéo
        float sliderVal = uiRenderer.getSliderVolumeAtCursor(screenX, screenY, screenWidth, screenHeight);
        if (sliderVal >= 0.0f && audioManager != null) {
            isDraggingVolumeSlider = true;
            audioManager.setVolume(sliderVal);
        }

    /**
     * Cập nhật tiến trình đào block khi đang giữ chuột trái.
     * Tích lũy thời gian đào ({@code miningElapsedTime}) cho đến khi đạt ngưỡng
     * Độ cứng ({@code hardness}) của block mục tiêu, lúc đó sẽ phá hủy block
     * và sinh {@code ItemEntity} tại vị trí đó.
     *
     * @param currentHit kết quả Raycast từ mắt Camera đến block đang ngắm
     * @param deltaTime  khoảng thời gian từ khung hình trước (giây)
     * @return tỷ lệ tiến trình đào (0.0f đến 1.0f), hoặc 0.0f nếu không đang đào
     */
    private float updateMiningProgress(Raycast.HitResult currentHit, float deltaTime) {
        // Nếu không giữ chuột trái hoặc không có block mục tiêu -> reset
        if (!isLeftMouseDown || currentHit == null || !currentHit.isHit()) {
            resetMiningProgress();
            return 0.0f;
        }

        // Lấy loại block đang nhắm
        Block targetBlock = currentHit.getBlock();

        // Block không thể phá hủy (hardness < 0) như Air, Bedrock, Water, Lava -> reset
        if (!targetBlock.isBreakable()) {
            resetMiningProgress();
            return 0.0f;
        }

        // Lấy tọa độ ô block đang nhắm
        int hitX = currentHit.getBlockX();
        int hitY = currentHit.getBlockY();
        int hitZ = currentHit.getBlockZ();

        // Nếu chuyển sang nhắm block khác -> reset tiến trình và bắt đầu lại từ 0
        if (hitX != miningTargetX || hitY != miningTargetY || hitZ != miningTargetZ) {
            miningTargetX = hitX;
            miningTargetY = hitY;
            miningTargetZ = hitZ;
            miningElapsedTime = 0.0f;
        }

        // Tích lũy thời gian đào
        miningElapsedTime += deltaTime;
        float requiredTime = targetBlock.getHardness();

        // Nếu đã đào đủ thời gian -> phá hủy block, sinh item rơi và reset tiến trình
        if (miningElapsedTime >= requiredTime) {
            world.breakBlockAndSpawnDrop(miningTargetX, miningTargetY, miningTargetZ);
            resetMiningProgress();
            return 0.0f;
        }

        // Trả về tỷ lệ tiến trình đào (giới hạn trong [0.0, 1.0])
        return Math.min(1.0f, miningElapsedTime / requiredTime);
    }

    /**
     * Đặt lại trạng thái đào block về mặc định (không có mục tiêu, thời gian = 0).
     * Được gọi khi: nhả chuột trái, chuyển sang block khác, block không thể đào,
     * hoặc đào xong phá hủy block thành công.
     */
    private void resetMiningProgress() {
        miningTargetX = Integer.MIN_VALUE;
        miningTargetY = Integer.MIN_VALUE;
        miningTargetZ = Integer.MIN_VALUE;
        miningElapsedTime = 0.0f;
    }

    /**
     * Xử lý sự kiện click chuột phải: Đặt block từ Hotbar vào thế giới.
     * - Lấy block đang chọn trong Hotbar
     * - Bắn tia Raycast tìm mặt phẳng đang nhìn
     * - Đặt block vào tọa độ trước mặt block bị tia chạm (prevBlock)
     * - Kiểm tra không trùng vị trí người chơi (AABB) trước khi đặt
     */
    private void handleRightClickPlaceBlock() {
        if (world == null || camera == null || player == null) {
            return;
        }

        // Lấy Hotbar của Player thông qua Composition
        Hotbar hotbar = player.getHotbar();

        // Kiểm tra ô đang chọn có block khả dụng không
        Block selectedBlock = hotbar.getSelectedBlock();
        if (selectedBlock == Block.AIR || hotbar.getSelectedCount() <= 0) {
            return;
        }

        // Bắn tia từ mắt Camera theo hướng nhìn, tầm với 5.0 ô voxel
        Raycast.HitResult hit = Raycast.cast(world, camera.getPosition(), camera.getFront(), RAYCAST_REACH);
        if (!hit.isHit()) {
            return;
        }

        // Vị trí đặt block = tọa độ ô trước mặt block bị tia chạm (prevBlock)
        int placeX = hit.getPrevBlockX();
        int placeY = hit.getPrevBlockY();
        int placeZ = hit.getPrevBlockZ();

        // Chỉ đặt nếu tọa độ nằm trong biên thế giới VÀ không trùng hộp va chạm người chơi
        if (world.isInsideWorld(placeX, placeY, placeZ) && !player.intersectsBlock(placeX, placeY, placeZ)) {
            Block consumed = hotbar.consumeSelectedItem();
            if (consumed != Block.AIR) {
                world.setBlock(placeX, placeY, placeZ, consumed.getId());
            }
        }
    }

    // CALLBACK RESIZE CỬA SỔ
    // =====================================================================

    /**
     * Xử lý khi cửa sổ bị thay đổi kích thước.

    @Override

    /**
     * Xử lý sự kiện phím bàn phím được nhấn xuống.
     * - Phím ESC: Tạm dừng game (lần 1) hoặc thoát hoàn toàn (lần 2 khi đang Pause)
     * - Phím 1-9: Chọn ô Hotbar tương ứng
     *
     * @param keycode mã phím libGDX (hằng số {@link Input.Keys})
     * @return true nếu sự kiện đã được xử lý
     */
    @Override
    public boolean keyDown(int keycode) {
        // Phím ESC: Tạm dừng (lần 1) hoặc thoát game (lần 2 khi đang Pause)
        if (keycode == Input.Keys.ESCAPE) {
            if (!isPaused) {
                pauseGame();
            } else {
                // ESC lần 2 khi đang Pause -> đóng cửa sổ, thoát hoàn toàn khỏi game
                Gdx.app.exit();
            }
            return true;
        }

        // Phím số 1..9: Chọn ô trên thanh 9 ô Hotbar (chỉ khi đang chơi, không Pause)
        if (!isPaused && player != null) {
            int slotIndex = getHotbarSlotFromKeycode(keycode);
            if (slotIndex >= 0) {
                player.getHotbar().selectSlot(slotIndex);
                return true;
            }
        }

        return false;
    }

    /**
     * Chuyển đổi mã phím libGDX sang chỉ số ô Hotbar (0-8).
     * Phím 1 -> ô 0, phím 2 -> ô 1, ..., phím 9 -> ô 8.
     *
     * @param keycode mã phím libGDX
     * @return chỉ số ô Hotbar (0-8), hoặc -1 nếu phím không phải 1-9
     */
    private int getHotbarSlotFromKeycode(int keycode) {
        if (keycode >= Input.Keys.NUM_1 && keycode <= Input.Keys.NUM_9) {
            return keycode - Input.Keys.NUM_1;
        }
        return -1;
    }

    @Override

    /**
     * Xử lý sự kiện nhấn chuột (Mouse Button Down).

     * - Khi đang chơi: Chuột trái bắt đầu đào block, chuột phải đặt block
     *
     * @param screenX tọa độ X chuột trên màn hình (pixel, từ trái sang phải)

    @Override
    public boolean touchDown(int screenX, int screenY, int pointer, int button) {

        currentMouseX = screenX;
        currentMouseY = screenY;

        if (isPaused) {

        if (button == Input.Buttons.LEFT) {
            isLeftMouseDown = true;
        } else if (button == Input.Buttons.RIGHT) {
            handleRightClickPlaceBlock();
        }

        return true;

     * @param screenY tọa độ Y chuột trên màn hình

     * @param button  nút chuột
     * @return true nếu sự kiện đã được xử lý

    @Override
    public boolean touchUp(int screenX, int screenY, int pointer, int button) {
        currentMouseX = screenX;
        currentMouseY = screenY;

        if (isPaused) {
            if (button == Input.Buttons.LEFT) {
                handlePauseMenuMouseClick(screenX, screenY, false);
            }
            return true;
        }

        // Nhả chuột trái: dừng đào block và reset tiến trình
        if (button == Input.Buttons.LEFT) {
            isLeftMouseDown = false;
            resetMiningProgress();
        }

        return true;
    }

    /**
     * Xử lý sự kiện kéo chuột khi đang giữ nút (Mouse Drag / touchDragged).
     * Khi đang Pause và đang kéo thanh trượt Volume: cập nhật âm lượng theo vị trí chuột.
     *
     * @param screenX tọa độ X chuột trên màn hình
     * @param screenY tọa độ Y chuột trên màn hình
     * @param pointer chỉ số con trỏ
     * @return true nếu sự kiện đã được xử lý
     */
    @Override
    public boolean touchDragged(int screenX, int screenY, int pointer) {
        currentMouseX = screenX;
        currentMouseY = screenY;

        // Khi đang Pause và đang kéo thanh trượt Volume: cập nhật âm lượng mượt mà theo vị trí chuột
        if (isPaused && isDraggingVolumeSlider && uiRenderer != null && audioManager != null) {
            int screenWidth = Gdx.graphics.getWidth();
            int screenHeight = Gdx.graphics.getHeight();
            float newVol = uiRenderer.getSliderVolumeAtCursor(screenX, screenY, screenWidth, screenHeight);
            if (newVol >= 0.0f) {
                audioManager.setVolume(newVol);
            }
        }

        return true;
    }

    /**
     * Xử lý sự kiện di chuyển chuột khi không giữ nút (Mouse Move / mouseMoved).
     * Cập nhật tọa độ chuột hiện tại để dùng cho Hit-Testing hover trên Pause Menu.
     * Việc xoay Camera được xử lý riêng trong {@link #processMouseLook()}.
     *
     * @param screenX tọa độ X chuột trên màn hình
     * @param screenY tọa độ Y chuột trên màn hình
     * @return true nếu sự kiện đã được xử lý
     */
    @Override
    public boolean mouseMoved(int screenX, int screenY) {
        currentMouseX = screenX;
        currentMouseY = screenY;
        return true;
    }

    /**
     * Xử lý sự kiện cuộn con lăn chuột (Mouse Scroll Wheel).
     * Cuộn lên/xuống để chuyển đổi giữa 9 ô trên thanh Hotbar (xoay vòng modulo 9).
     *
     * @param amountX lượng cuộn theo trục X (thường = 0)
     * @param amountY lượng cuộn theo trục Y (âm = cuộn lên, dương = cuộn xuống)
     * @return true nếu sự kiện đã được xử lý
     */
    @Override
    public boolean scrolled(float amountX, float amountY) {
        if (!isPaused && player != null) {
            // amountY > 0 (cuộn xuống) = tăng chỉ số ô -> truyền dương
            // amountY < 0 (cuộn lên)   = giảm chỉ số ô -> truyền âm
            // Hàm hotbar.scroll() kỳ vọng: giá trị dương = cuộn lên (giảm slot),
            // giá trị âm = cuộn xuống (tăng slot) -> đảo dấu amountY
            player.getHotbar().scroll(-amountY);
        }
        return true;
    }

    /**
     * Xử lý sự kiện hủy bỏ thao tác chạm (touchCancelled).
     * Trên Desktop sự kiện này hiếm khi xảy ra. Xử lý tương tự nhả chuột.
     *
     * @param screenX tọa độ X
     * @param screenY tọa độ Y
     * @param pointer chỉ số con trỏ
     * @param button  nút chuột
     * @return true
     */
    @Override
    public boolean touchCancelled(int screenX, int screenY, int pointer, int button) {
        return touchUp(screenX, screenY, pointer, button);
    }

### 4. Gameplay & Hệ Thống Điều Khiển
- **Di chuyển & Xoay góc nhìn:** `W, A, S, D` di chuyển 4 hướng trên mặt phẳng ngang XZ theo hướng `yaw`; di chuột xoay góc nhìn 360° với độ nhạy `MOUSE_SENSITIVITY = 0.12f` và giới hạn `pitch` trong `[-89.0°, 89.0°]`[cite: 1].
- **Nhảy & Bơi (`Space` 1 lần):** Nhảy khi `onGround == true` (`JUMP_VELOCITY = 8.6f`)[cite: 1]; khi đang trong Nước hoặc Lava sẽ bơi nổi lên với vận tốc dọc `4.2f`[cite: 1].
- **Chế độ Bay (Double-tap `Space` trong `<= 0.30s`):** Bật/tắt Fly Mode, tắt trọng lực; giữ `Space` bay lên với `FLY_ASCEND_SPEED = 7.0f` (nhả `Space` thì `velocity.y = 0`); tốc độ ngang `1.8x` (thêm `1.35x` nếu giữ `Left Ctrl`); chạm mặt đất sẽ tự thoát Fly Mode[cite: 1].
- **Rón rén & Thoát Bay (`Left Shift`):**
  - Khi đang Bay: thoát Fly Mode ngay lập tức, bật lại trọng lực[cite: 1].
  - Khi đứng/đi trên mặt đất: bật **Sneak**, giảm tốc độ còn `0.5x`, nội suy mượt tầm mắt Camera từ `1.62f` xuống `1.45f` (tốc độ nội suy `deltaTime * 14.0f`), và kích hoạt **giữ mép block** (`preventFallingOffEdgeX/Z` theo bước `0.02f`) không cho rơi xuống vực[cite: 1].
- **Chạy nhanh (`Left Ctrl`):** Bật **Sprint** khi đang di chuyển ngang và không Sneak; tăng `1.5x` tốc độ đi bộ kèm hiệu ứng mở rộng Dynamic FOV[cite: 1].
- **9 Ô Hotbar (`1`–`9` hoặc Cuộn chuột):**
  - `SLOT_COUNT = 9`, `MAX_STACK_SIZE = 64`[cite: 1].
  - Cuộn chuột lên giảm chỉ số ô, cuộn xuống tăng chỉ số ô (xoay vòng modulo 9)[cite: 1].
  - Trang bị khởi đầu: Ô 1 `Dirt x16`, Ô 2 `Grass Block x16`, Ô 3 `Stone x16`, Ô 4 `Oak Log x12`, Ô 5 `Oak Leaves x16`, Ô 6–9 trống[cite: 1].
  - Cơ chế nhặt đồ (`addItem`): ưu tiên cộng dồn vào các ô có sẵn cùng loại chưa đầy 64, sau đó mới xếp vào ô trống đầu tiên từ trái sang phải[cite: 1].
- **Đào Block (Giữ Chuột Trái):**
  - Tầm với `RAYCAST_REACH = 5.0f`[cite: 1].
  - Tích lũy `miningElapsedTime` cho đến khi `>= hardness` sẽ đổi ô thành `Block.AIR` và gọi `target.onBreak(...)` sinh `ItemEntity` tại tâm block `(x + 0.5, y + 0.5, z + 0.5)`[cite: 1].
  - Nhả chuột, nhìn sang ô khác, hoặc ngắm vào khối không phá được sẽ reset tiến trình về `0`[cite: 1].
- **Đặt Block (Click Chuột Phải):**
  - Lấy block ở ô Hotbar đang chọn (nếu khác `AIR` và `count > 0`), đặt vào tọa độ `(prevBlockX, prevBlockY, prevBlockZ)` của Raycast nếu nằm trong biên thế giới và `!player.intersectsBlock(...)`, đồng thời trừ 1 item khỏi ô đang chọn[cite: 1].
- **Tạm dừng & Thoát (`ESC`):**
  - Nhấn `ESC` lần 1: bật `isPaused`, reset tiến trình đào, mở con trỏ chuột (`GLFW_CURSOR_NORMAL`), đóng băng cập nhật thực thể[cite: 1].
  - Nhấn `ESC` lần 2 khi đang Pause: đặt cờ đóng cửa sổ và thoát hoàn toàn khỏi game[cite: 1].

### 6. Đồ Họa, Shader, Texture & Hiệu Năng 60 FPS
- **OpenGL & Vòng lặp:** OpenGL 3.3 Core Profile trong backend LWJGL 3 của libGDX, bật `GL_DEPTH_TEST`, `GL_CULL_FACE` (`GL_BACK`, `GL_CCW`), bật V-Sync `config.useVsync(true)` đạt 60 FPS mượt mà, giới hạn `deltaTime <= 0.05f`.
- **Màu trời, Sương mù & Ống kính:**
  - `SKY_COLOR = (0.53f, 0.81f, 0.92f)` dùng chung cho màu xóa nền và `uFogColor`.
  - Sương mù khoảng cách tuyến tính trong View Space: `uFogStart = 28.0f`, `uFogEnd = 58.0f`.
  - Camera: `BASE_FOV_DEG = 70.0f` (`+8.0f` khi Sprint, `+5.0f` khi Fly, nội suy `deltaTime * 10.0f`), `NEAR_PLANE = 0.08f`, `FAR_PLANE = 160.0f`, ma trận tính qua `Matrix4.setToProjection` và `setToLookAt`.
- **Định dạng Đỉnh & Shader (`ShaderProgram`):**
  - 6 floats/đỉnh: `[x, y, z, u, v, shade]` (`location 0: vec3 aPos`, `location 1: vec2 aTexCoord`, `location 2: float aShade`) quản lý qua `VertexAttributes` của libGDX.
  - Fragment Shader: chọn màu từ `uTexture * uTintColor` (`uUseTexture == 1`) hoặc `uTintColor` (`uUseTexture == 0`); loại bỏ pixel trong suốt (`alpha < 0.05 -> discard`); nhân `vShade`; hòa trộn sương mù nếu `uUseFog == 1`.
  - Xóa tiền tố `prependVertexCode`/`prependFragmentCode` để giữ `#version 330 core` ở dòng đầu tiên.
- **Ánh sáng & Face Culling (`Chunk`):**
  - Directional Face Shading: `SHADE_TOP = 1.00f`, `SHADE_BOTTOM = 0.55f`, `SHADE_NORTH_SOUTH = 0.85f`, `SHADE_EAST_WEST = 0.72f`.
  - Face Culling: Khối lỏng chỉ vẽ mặt tiếp giáp `Block.AIR`; khối rắn chỉ vẽ mặt tiếp giáp `!neighbor.isSolid()`.
  - Kết xuất thông qua `com.badlogic.gdx.graphics.Mesh`.
  - Khi `World.setBlock(...)` sửa block ở biên Chunk (`0` hoặc `15`), tự động `markDirty()` các Chunk lân cận và chỉ rebuild những Chunk có `dirty == true`.
- **Texture Atlas (`Texture`):**
  - Lưới `4 x 4` tile, mỗi tile `16 x 16` pixel -> Atlas `64 x 64` RGBA, lọc `TextureFilter.Nearest`, bao biên `TextureWrap.ClampToEdge`, lùi biên UV `eps = 0.0008f` chống Texture Bleeding.
  - Thứ tự 10 tile đang dùng: `0=DIRT`, `1=GRASS_TOP`, `2=GRASS_SIDE`, `3=STONE`, `4=BEDROCK`, `5=OAK_LOG_SIDE`, `6=OAK_LOG_TOP`, `7=OAK_LEAVES`, `8=WATER`, `9=LAVA`.
  - Xử lý ảnh thông minh bằng `Pixmap`: tự tách sheet `32x32` của `grass-block.png` thành 3 tile (`GRASS_TOP`, `GRASS_SIDE`, `DIRT`); cắt khung vuông đầu tiên của `water_still.png` và nhuộm xanh nước biển (`alpha = 225`) nếu ảnh nguồn là thang xám; có đầy đủ 10 tile Procedural Pixel Art (seed `20261114L`) làm nền dự phòng; tự động `dispose()` Pixmap sau khi nạp lên GPU.
- **Tối ưu Hiệu năng & Quản lý Bộ nhớ:**
  - Tái sử dụng các `Matrix4` làm trường thành viên để tránh sinh rác GC trong Game Loop.
  - Dùng chung `Mesh` theo loại `Block` cho mọi `ItemEntity` (`SHARED_MESHES`).
  - Giải phóng triệt để tài nguyên GPU qua `Disposable.dispose()` cho mọi lớp `Chunk`, `World`, `Texture`, `ShaderProgram`, `ItemEntity`.

### 7. Giao Diện Người Dùng (UI/HUD) & Font 5x7 (`UIRenderer`)

- **Overlay 3D quanh Block mục tiêu:**
